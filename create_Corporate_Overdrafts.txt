import re
import numpy as np
import pandas as pd
import dataiku
from dataiku.core.sql import SQLExecutor2

PROJECT = 'VALIDATIONEWSCORPORATEPORTFOLIODAILY'
CONNECTION = 'DEV_AA_TEAMS'
SOURCE = 'Overdraft_Excess_s'
OUTPUT = 'overdraft_features_Corporate'
TRAINING = 'EWS_training_Corporate'
# Candidate assumption: the source is an exhaustive extract of qualifying excesses.
# Absence is zero only within covered months and stable, unambiguous customer mappings.
MIN_WEEKDAY_COVERAGE = 0.80
DURATIONS = (5, 10, 15)
project = dataiku.api_client().get_project(PROJECT)
snow = SQLExecutor2(connection=CONNECTION)
ora = SQLExecutor2(connection='PRD_BDWH')

def query(exe, sql):
    x = exe.query_to_df(sql)
    x.columns = x.columns.str.upper()
    return x

def physical(name):
    p = project.get_dataset(name).get_settings().get_raw()['params']
    assert p['connection'] == CONNECTION, name
    def q(v):
        v = str(v).replace('${projectKey}', PROJECT)
        assert re.fullmatch(r'[A-Za-z_][A-Za-z_0-9$]*', v), v
        return '"'+v+'"'
    return q(p['schema'])+'.'+q(p['table'])

def norm(s):
    return s.astype('string').str.strip().str.replace(r'^(\d+)\.0+$',r'\1',regex=True).replace('',pd.NA)

def dates(s):
    return pd.to_datetime(s, errors='coerce', utc=True).dt.tz_localize(None).dt.normalize()

def fmt(d): return pd.Timestamp(d).strftime('%Y-%m-%d')

def write_blocked(reason):
    # Explicit non-observation; never joins the backtest.
    x = pd.DataFrame({'ENTITY_ID':['__NO_OBSERVATION__'], 'MONTH':[pd.Timestamp('1900-01-01')],
                      'AS_OF_DATE':[pd.Timestamp('1900-01-01')], 'THRESHOLD_A':[5],
                      'N_LATEST_12M':[0], 'N_PREVIOUS_12M':[0], 'DELTA_EPISODES':[0],
                      'PREV_DELTA_EPISODES':[0], 'VALID':[0], 'FEATURE_STATUS':[reason]})
    dataiku.Dataset(OUTPUT,project_key=PROJECT).write_with_schema(x)
    print('BLOCKED |',reason,flush=True)

def main():
    cols=['ACCOUNT_ID','ACCOUNT_SUBTYPE','PAST_DUE_AMOUNT_CTV','DAYS_PAST_DUE','CUSTOMER','MAP_DTE']
    ds=dataiku.Dataset(SOURCE,project_key=PROJECT)
    actual={c['name'].upper():c['name'] for c in ds.read_schema()}
    assert set(cols)<=set(actual), 'Missing source columns: '+str(set(cols)-set(actual))
    p=project.get_dataset(SOURCE).get_settings().get_raw().get('params',{})
    if p.get('connection')==CONNECTION:
        qcols=','.join('"'+actual[c]+'"' for c in cols)
        # Prefix account/customer strings to prevent SQL driver numeric inference.
        qcols=','.join(("'K|'||TRIM(TO_VARCHAR(\""+actual[c]+"\")) AS "+c) if c in ('ACCOUNT_ID','CUSTOMER') else '"'+actual[c]+'" AS '+c for c in cols)
        x=query(snow,'SELECT DISTINCT '+qcols+' FROM '+physical(SOURCE))
        for c in ('ACCOUNT_ID','CUSTOMER'): x[c]=x[c].astype('string').str.slice(2)
    else:
        x=ds.get_dataframe(columns=[actual[c] for c in cols],infer_with_pandas=False)
        x.columns=x.columns.str.upper()
    if x.empty: return write_blocked('EMPTY_SOURCE')
    raw=len(x)
    for c in ['ACCOUNT_ID','CUSTOMER']: x[c]=norm(x[c])
    x['MAP_DTE']=dates(x.MAP_DTE)
    x['DAYS_PAST_DUE']=pd.to_numeric(x.DAYS_PAST_DUE,errors='coerce')
    x['PAST_DUE_AMOUNT_CTV']=pd.to_numeric(x.PAST_DUE_AMOUNT_CTV,errors='coerce')
    print('SOURCE QA | rows',raw,'| subtypes',x.ACCOUNT_SUBTYPE.nunique(),flush=True)
    print(x.groupby('ACCOUNT_SUBTYPE',dropna=False).agg(N_ROWS=('ACCOUNT_ID','size'),N_ACCOUNTS=('ACCOUNT_ID','nunique'),N_CUSTOMERS=('CUSTOMER','nunique')).to_string(),flush=True)
    invalid=x[['ACCOUNT_ID','CUSTOMER','MAP_DTE','DAYS_PAST_DUE']].isna().any(axis=1)|x.DAYS_PAST_DUE.lt(0)|x.DAYS_PAST_DUE.mod(1).ne(0)|x.DAYS_PAST_DUE.gt(36500)
    if invalid.any(): return write_blocked('INVALID_ACCOUNT_CUSTOMER_DATE_OR_DPD')
    # Identical observations may repeat. Conflicting account-day DPD/owner is not resolved arbitrarily.
    x=x[['ACCOUNT_ID','CUSTOMER','MAP_DTE','DAYS_PAST_DUE']].drop_duplicates()
    if x.duplicated(['ACCOUNT_ID','MAP_DTE']).any():
        print('CONFLICTING ACCOUNT-DAYS',int(x.duplicated(['ACCOUNT_ID','MAP_DTE'],keep=False).sum()),flush=True)
        return write_blocked('CONFLICTING_ACCOUNT_DAY')
    x['MONTH']=x.MAP_DTE.dt.to_period('M').dt.to_timestamp()
    lo=x.MAP_DTE.min(); hi=x.MAP_DTE.max()
    t=physical(TRAINING)
    bounds=query(snow,'SELECT MIN(TO_DATE(MAP_DTE)) START_DATE,MAX(TO_DATE(MAP_DTE)) END_DATE FROM '+t)
    train_end=pd.Timestamp(bounds.END_DATE.iloc[0]).normalize().tz_localize(None)
    # Drop partial boundary months. First/last weekday proxy is a conservative completeness check.
    months=pd.date_range(lo.to_period('M').start_time,min(hi,train_end).to_period('M').start_time,freq='MS',name='MONTH')
    observed=x[['MAP_DTE','MONTH']].drop_duplicates()
    covered=[]
    for m in months:
        end=m+pd.offsets.MonthEnd(0)
        weekdays=pd.bdate_range(m,end)
        seen=set(observed.loc[observed.MONTH.eq(m),'MAP_DTE'])
        n=sum(d in seen for d in weekdays)
        good=bool(lo<=weekdays[0] and hi>=weekdays[-1] and end<train_end and n/len(weekdays)>=MIN_WEEKDAY_COVERAGE)
        covered.append(good)
        print('MONTH COVERAGE |',fmt(m),'| dates',len(seen),'| weekdays',n,'/',len(weekdays),'| candidate complete',good,flush=True)
    coverage=pd.Series(covered,index=months,dtype=bool)
    # A crossing needs 24 complete months plus the immediately preceding state.
    if len(months)<25 or not coverage.astype(int).rolling(25).sum().eq(25).any():
        return write_blocked('INSUFFICIENT_25_CONSECUTIVE_MONTHS')
    # Independent account episodes: a falling DPD, zero, or implied new start
    # after the preceding observation indicates a reset. Missing rows alone do not.
    x=x.sort_values(['ACCOUNT_ID','MAP_DTE']).reset_index(drop=True)
    g=x.groupby('ACCOUNT_ID',sort=False)
    prev_dpd=g.DAYS_PAST_DUE.shift(); prev_date=g.MAP_DTE.shift()
    inferred=x.MAP_DTE-pd.to_timedelta(x.DAYS_PAST_DUE.clip(lower=1)-1,unit='D')
    new=(prev_date.isna()|prev_dpd.eq(0)|x.DAYS_PAST_DUE.lt(prev_dpd)|inferred.gt(prev_date)) & x.DAYS_PAST_DUE.gt(0)
    x['EPISODE']=new.groupby(x.ACCOUNT_ID).cumsum().astype('int64')
    x['START_ESTIMATE']=inferred
    positive=x[x.DAYS_PAST_DUE.gt(0)].copy()
    # At first observation of each episode, record whether its start predates the extract.
    first=positive.drop_duplicates(['ACCOUNT_ID','EPISODE'])[['ACCOUNT_ID','EPISODE','START_ESTIMATE']].rename(columns={'START_ESTIMATE':'FIRST_START_ESTIMATE'})
    positive=positive.merge(first,on=['ACCOUNT_ID','EPISODE'],how='left',validate='many_to_one')
    positive=positive[positive.FIRST_START_ESTIMATE.ge(lo)]
    events=[]
    for d in DURATIONS:
        a=positive[positive.DAYS_PAST_DUE.gt(d)].drop_duplicates(['ACCOUNT_ID','EPISODE'],keep='first').copy()
        a['THRESHOLD_A']=d
        events.append(a[['ACCOUNT_ID','CUSTOMER','MONTH','MAP_DTE','THRESHOLD_A']])
    events=pd.concat(events,ignore_index=True)
    print('EPISODES | first observed qualification by DPD',events.groupby('THRESHOLD_A').size().to_dict(),flush=True)
    # All Corporate entities in the training population; not only customers with observed excesses.
    universe=query(snow,"SELECT DISTINCT 'E|'||TRIM(TO_VARCHAR(ENTITY_ID)) ENTITY_KEY FROM "+t+" WHERE ENTITY_ID IS NOT NULL")
    corporate=set(norm(universe.ENTITY_KEY.astype('string').str.slice(2)).dropna())
    # Use a unique customer-to-entity mapping from the latest source snapshot in each month.
    # At the evaluation date (next month), the mapping snapshot is already in the past.
    parts=[]; counts=[]
    for m in months:
        if not coverage.loc[m]: continue
        end=m+pd.offsets.MonthBegin(1)
        sql=f"""WITH S AS (
          SELECT MAX(MAP_DTE) D FROM BDL_CREDIT_INSIGHT_PLY.ENTITY_CUSTOMER_MON_VW
          WHERE MAP_DTE>=DATE '{fmt(m)}' AND MAP_DTE<DATE '{fmt(end)}'
        ) SELECT DISTINCT 'E|'||TRIM(TO_CHAR(A.ENTITY_ID)) ENTITY_KEY,
          'C|'||TRIM(TO_CHAR(A.CUSTOMER_ID)) CUSTOMER_KEY
        FROM BDL_CREDIT_INSIGHT_PLY.ENTITY_CUSTOMER_MON_VW A JOIN S ON A.MAP_DTE=S.D
        WHERE A.ENTITY_ID IS NOT NULL AND A.CUSTOMER_ID IS NOT NULL"""
        mp=query(ora,sql)
        if mp.empty:
            print('MAPPING | missing month',fmt(m),flush=True);continue
        mp['ENTITY_ID']=norm(mp.ENTITY_KEY.astype('string').str.slice(2))
        mp['CUSTOMER']=norm(mp.CUSTOMER_KEY.astype('string').str.slice(2))
        mp=mp[['ENTITY_ID','CUSTOMER']].dropna().drop_duplicates()
        amb=mp.CUSTOMER.duplicated(keep=False)
        bad_entities=set(mp.loc[amb,'ENTITY_ID'])
        mp=mp.loc[~amb & mp.ENTITY_ID.isin(corporate) & ~mp.ENTITY_ID.isin(bad_entities)].copy()
        if mp.empty: continue
        # Stable mapped customer set required throughout both windows and previous state.
        sets=mp.groupby('ENTITY_ID').CUSTOMER.agg(lambda s:'|'.join(sorted(s))).rename('CUSTOMER_SET').reset_index()
        sets['MONTH']=m;parts.append(sets)
        a=events.loc[events.MONTH.eq(m)].merge(mp,on='CUSTOMER',how='left',validate='many_to_one')
        print('MAPPING |',fmt(m),'| entities',len(sets),'| event rows',len(a),'| mapped to Corporate',int(a.ENTITY_ID.notna().sum()),'| ambiguous customer rows',int(amb.sum()),flush=True)
        # Count independent account episodes, including multiple accounts of one entity.
        b=a.dropna(subset=['ENTITY_ID']).groupby(['ENTITY_ID','THRESHOLD_A']).size().rename('N_EPISODES').reset_index()
        b['MONTH']=m;counts.append(b)
    if not parts: return write_blocked('NO_CUSTOMER_ENTITY_MAPPING')
    p=pd.concat(parts,ignore_index=True)
    w=p.pivot(index='ENTITY_ID',columns='MONTH',values='CUSTOMER_SET').reindex(columns=months)
    # Equal consecutive customer sets; missing mapping invalidates the calendar window.
    stable=(w.notna() & w.eq(w.shift(1,axis=1))).fillna(False)
    valid25=stable.T.astype('int8').rolling(24,min_periods=24).sum().T.eq(24)
    if not valid25.any().any(): return write_blocked('NO_STABLE_25_MONTH_ENTITY_HISTORY')
    c=pd.concat(counts,ignore_index=True) if counts else pd.DataFrame(columns=['ENTITY_ID','THRESHOLD_A','MONTH','N_EPISODES'])
    out=[]
    for d in DURATIONS:
        z=c.loc[c.THRESHOLD_A.eq(d)]
        n=z.pivot(index='ENTITY_ID',columns='MONTH',values='N_EPISODES').reindex(index=w.index,columns=months).fillna(0).where(w.notna())
        latest=n.T.rolling(12,min_periods=12).sum().T; previous=latest.shift(12,axis=1)
        delta=latest-previous; prev_delta=delta.shift(1,axis=1)
        ok=valid25 & delta.notna() & prev_delta.notna()
        long=delta.where(ok).stack().rename('DELTA_EPISODES').reset_index()
        if long.empty: continue
        for name,mat in [('N_LATEST_12M',latest),('N_PREVIOUS_12M',previous),('PREV_DELTA_EPISODES',prev_delta)]:
            v=mat.where(ok).stack().rename(name).reset_index()
            long=long.merge(v,on=['ENTITY_ID','MONTH'],validate='one_to_one')
        long['AS_OF_DATE']=long.MONTH+pd.offsets.MonthBegin(1)
        long['THRESHOLD_A']=d;long['VALID']=1
        long['FEATURE_STATUS']='CANDIDATE_FILTERED_SOURCE'
        for col in ['DELTA_EPISODES','N_LATEST_12M','N_PREVIOUS_12M','PREV_DELTA_EPISODES']: long[col]=long[col].astype('int64')
        out.append(long)
    if not out: return write_blocked('NO_VALID_CONSECUTIVE_WINDOWS')
    f=pd.concat(out,ignore_index=True)
    assert not f.duplicated(['ENTITY_ID','MONTH','THRESHOLD_A']).any()
    assert f.DELTA_EPISODES.eq(f.N_LATEST_12M-f.N_PREVIOUS_12M).all()
    assert f.N_LATEST_12M.ge(0).all() and f.N_PREVIOUS_12M.ge(0).all()
    f['ENTITY_ID']=f.ENTITY_ID.astype(str)
    print('FEATURE QA |',len(f),'rows',f.ENTITY_ID.nunique(),'entities',flush=True)
    print(f.groupby(['MONTH','THRESHOLD_A']).agg(N_EVALUATIONS=('ENTITY_ID','size'),N_POSITIVE_DELTA=('DELTA_EPISODES',lambda s:int(s.gt(0).sum())),MAX_DELTA=('DELTA_EPISODES','max')).to_string(),flush=True)
    dataiku.Dataset(OUTPUT,project_key=PROJECT).write_with_schema(f)
    print('FINISHED |',OUTPUT,flush=True)

main()
