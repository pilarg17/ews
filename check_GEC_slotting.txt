# Read-only GEC Slotting diagnostic. Run as ONE notebook cell.
import dataiku
from dataiku.core.sql import SQLExecutor2

PROJECT = 'VALIDATIONEWSCORPORATEPORTFOLIODAILY'
START, END = '2024-11-07', '2026-10-29'
project = dataiku.api_client().get_project(PROJECT)

def table(name):
    p = project.get_dataset(name).get_settings().get_raw()['params']
    if p.get('connection') != 'DEV_AA_TEAMS':
        raise RuntimeError('Unexpected connection for ' + name)
    return '"' + p['schema'] + '"."' + p['table'].replace('${projectKey}', PROJECT) + '"'

T = table('EWS_training_Corporate')
G = table('GEC_links_B01_daily')
B = table('EWS_training_Corporate_GEC_Rating_DPD_Arrears_B02')

print('START | GEC Slotting mapping and threshold diagnostics | READ ONLY', flush=True)
sql = f"""
WITH SLOT_RAW AS (
 SELECT TRIM(TO_VARCHAR(ENTITY_ID)) MEMBER_ID, TO_DATE(MAP_DTE) D,
        COUNT(*) N_ROWS,
        MAX(TRY_TO_DOUBLE(ACA_FINAL_RATING_WORST::VARCHAR)) WORST,
        MAX(IFF(TRY_TO_BOOLEAN(ACA_SLOT_GRADE_CONFLICT::VARCHAR)=FALSE,1,0)) NO_CONFLICT
 FROM {T}
 WHERE ACA_FINAL_RATING_WORST IS NOT NULL AND ENTITY_ID IS NOT NULL
   AND TO_DATE(MAP_DTE)<=DATE '{END}'
 GROUP BY 1,2
), SLOT AS (
 SELECT *, IFF(N_ROWS=1 AND WORST IN (1,2,3,4),WORST,NULL) GRADE
 FROM SLOT_RAW
), HIST AS (
 SELECT *,
 MAX(IFF(GRADE=1,D,NULL)) OVER(PARTITION BY MEMBER_ID ORDER BY D ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) L1,
 MAX(IFF(GRADE=2,D,NULL)) OVER(PARTITION BY MEMBER_ID ORDER BY D ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) L2,
 MAX(IFF(GRADE=3,D,NULL)) OVER(PARTITION BY MEMBER_ID ORDER BY D ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) L3,
 MAX(IFF(GRADE=4,D,NULL)) OVER(PARTITION BY MEMBER_ID ORDER BY D ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) L4
 FROM SLOT
), W AS (SELECT COLUMN1::INTEGER M FROM VALUES (3),(6),(12)),
 FEATURES AS (
 SELECT MEMBER_ID,D,GRADE,M,
 GRADE-LEAST(GRADE,CASE WHEN L1>=DATEADD('MONTH',-M,D) THEN 1
                       WHEN L2>=DATEADD('MONTH',-M,D) THEN 2
                       WHEN L3>=DATEADD('MONTH',-M,D) THEN 3
                       WHEN L4>=DATEADD('MONTH',-M,D) THEN 4 END) FEATURE
 FROM HIST CROSS JOIN W
), COHORT AS (SELECT DISTINCT MEMBER_ID FROM SLOT WHERE GRADE IS NOT NULL),
 MEMBER_T AS (
 SELECT TRIM(TO_VARCHAR(t.ENTITY_ID)) MEMBER_ID, TO_DATE(t.MAP_DTE) D,
        COUNT(*) N_TRAIN_ROWS
 FROM {T} t JOIN COHORT c ON c.MEMBER_ID=TRIM(TO_VARCHAR(t.ENTITY_ID))
 WHERE TO_DATE(t.MAP_DTE) BETWEEN DATE '{START}' AND DATE '{END}'
 GROUP BY 1,2
), LINKS AS (
 SELECT TO_DATE(MAP_DTE) D, TRIM(TO_VARCHAR(ENTITY_ID)) MEMBER_ID,
        MIN(TRIM(TO_VARCHAR(GEC_ID))) GEC_ID,
        COUNT(DISTINCT TRIM(TO_VARCHAR(GEC_ID))) N_GROUPS
 FROM {G} WHERE ENTITY_ID IS NOT NULL AND GEC_ID IS NOT NULL
   AND TO_DATE(MAP_DTE) BETWEEN DATE '{START}' AND DATE '{END}'
 GROUP BY 1,2
), MEMBER_LINK AS (
 SELECT t.*,l.GEC_ID FROM MEMBER_T t JOIN LINKS l USING (MEMBER_ID,D)
 WHERE l.N_GROUPS=1
), RELEVANT_GROUP AS (SELECT DISTINCT D,GEC_ID FROM MEMBER_LINK),
 GROUP_MEMBERS AS (
 SELECT l.D,l.GEC_ID,l.MEMBER_ID
 FROM LINKS l JOIN RELEVANT_GROUP g USING (D,GEC_ID)
 WHERE l.N_GROUPS=1
), GROUP_TRAIN AS (
 SELECT g.D,g.GEC_ID,g.MEMBER_ID,
        COUNT(t.ENTITY_ID) N_ROWS,
        COUNT(DISTINCT NULLIF(TRIM(TO_VARCHAR(t.GEC_HEAD_ID)),'')) N_HEADS,
        MIN(NULLIF(TRIM(TO_VARCHAR(t.GEC_HEAD_ID)),'')) HEAD_ID,
        COUNT_IF(t.GEC_HEAD_ID IS NULL) N_NULL_HEAD
 FROM GROUP_MEMBERS g LEFT JOIN {T} t
 ON TRIM(TO_VARCHAR(t.ENTITY_ID))=g.MEMBER_ID AND TO_DATE(t.MAP_DTE)=g.D
 GROUP BY 1,2,3
), GROUP_EVIDENCE AS (
 SELECT D,GEC_ID,MIN(HEAD_ID) HEAD_ID,
        COUNT(DISTINCT HEAD_ID) N_DISTINCT_HEADS,
        MAX(N_HEADS) MAX_MEMBER_HEADS,
        COUNT_IF(N_ROWS=0) N_ABSENT_MEMBERS,
        COUNT_IF(N_ROWS>0 AND (N_NULL_HEAD>0 OR HEAD_ID IS NULL)) N_MISSING_HEAD_MEMBERS
 FROM GROUP_TRAIN GROUP BY 1,2
), HEAD_CANDIDATES AS (
 SELECT * FROM GROUP_EVIDENCE WHERE N_DISTINCT_HEADS=1 AND MAX_MEMBER_HEADS=1
), HEAD_TRAIN AS (
 SELECT h.D,h.GEC_ID,h.HEAD_ID,COUNT(t.ENTITY_ID) N_HEAD_ROWS
 FROM HEAD_CANDIDATES h JOIN {T} t
 ON TRIM(TO_VARCHAR(t.ENTITY_ID))=h.HEAD_ID AND TO_DATE(t.MAP_DTE)=h.D
 GROUP BY 1,2,3
), MAPPED AS (
 SELECT l.D,l.MEMBER_ID,h.HEAD_ID,
        IFF(h.N_ABSENT_MEMBERS=0 AND h.N_MISSING_HEAD_MEMBERS=0,1,0) STRICT_OK,
        IFF(hl.N_GROUPS=1 AND hl.GEC_ID=l.GEC_ID,1,0) HEAD_LINK_OK,
        IFF(gt.HEAD_ID=h.HEAD_ID AND gt.N_HEADS=1,1,0) MEMBER_HEAD_OK
 FROM MEMBER_LINK l JOIN HEAD_CANDIDATES h USING (D,GEC_ID)
 JOIN HEAD_TRAIN ht ON ht.D=h.D AND ht.GEC_ID=h.GEC_ID AND ht.HEAD_ID=h.HEAD_ID
 LEFT JOIN LINKS hl ON hl.MEMBER_ID=h.HEAD_ID AND hl.D=h.D
 JOIN GROUP_TRAIN gt ON gt.D=l.D AND gt.GEC_ID=l.GEC_ID AND gt.MEMBER_ID=l.MEMBER_ID
 WHERE l.MEMBER_ID<>h.HEAD_ID AND ht.N_HEAD_ROWS>0
   AND l.N_TRAIN_ROWS=1
), REASON_ROWS AS (
 SELECT mt.D,mt.MEMBER_ID,
   CASE WHEN l.MEMBER_ID IS NULL THEN 'NO_GEC_LINK'
        WHEN l.N_GROUPS<>1 THEN 'AMBIGUOUS_GEC_LINK'
        WHEN COALESCE(ge.N_DISTINCT_HEADS,0)=0 THEN 'NO_HEAD_EVIDENCE'
        WHEN ge.N_DISTINCT_HEADS>1 OR ge.MAX_MEMBER_HEADS>1 THEN 'CONFLICTING_HEADS'
        WHEN ht.HEAD_ID IS NULL THEN 'HEAD_NOT_IN_TRAINING'
        WHEN ge.HEAD_ID=mt.MEMBER_ID THEN 'MEMBER_IS_HEAD'
        WHEN ge.N_ABSENT_MEMBERS>0 THEN 'STRICT_REJECT_MISSING_MEMBERS'
        WHEN ge.N_MISSING_HEAD_MEMBERS>0 THEN 'STRICT_REJECT_MISSING_HEAD_VALUES'
        ELSE 'STRICT_ACCEPTED' END REASON
 FROM MEMBER_T mt LEFT JOIN LINKS l ON l.MEMBER_ID=mt.MEMBER_ID AND l.D=mt.D
 LEFT JOIN GROUP_EVIDENCE ge ON ge.D=mt.D AND ge.GEC_ID=l.GEC_ID
 LEFT JOIN HEAD_TRAIN ht ON ht.D=ge.D AND ht.GEC_ID=ge.GEC_ID AND ht.HEAD_ID=ge.HEAD_ID
 WHERE mt.N_TRAIN_ROWS=1
), STAGES AS (
 SELECT '01_ALL_SLOT_MEMBERS' STAGE,D,MEMBER_ID,'-' HEAD_ID FROM MEMBER_T WHERE N_TRAIN_ROWS=1
 UNION ALL SELECT '02_UNAMBIGUOUS_GEC_LINK',D,MEMBER_ID,'-' FROM MEMBER_LINK WHERE N_TRAIN_ROWS=1
 UNION ALL SELECT '03_HEAD_CONSENSUS',D,MEMBER_ID,HEAD_ID FROM MAPPED
 UNION ALL SELECT '04_HEAD_AND_MEMBER_CONFIRMED',D,MEMBER_ID,HEAD_ID FROM MAPPED WHERE HEAD_LINK_OK=1 AND MEMBER_HEAD_OK=1
 UNION ALL SELECT '05_CURRENT_STRICT_MAPPING',D,MEMBER_ID,HEAD_ID FROM MAPPED WHERE STRICT_OK=1
), HEAD_KEYS AS (SELECT DISTINCT D,HEAD_ID FROM MAPPED),
 HEAD_ELIG AS (
 SELECT k.D,k.HEAD_ID,
        MAX(IFF(TRY_TO_NUMBER(t.TGT_ANY_SAMEDAY::VARCHAR)=0
          AND TRY_TO_NUMBER(b.HAS_VALID_ENTITY_HISTORY::VARCHAR)=1
          AND TRY_TO_NUMBER(b.LOAN_COUNT::VARCHAR)>0
          AND TRY_TO_NUMBER(b.LOAN_SOURCE_DUPLICATES::VARCHAR)=0
          AND TRY_TO_NUMBER(b.NOMINAL_KNOWN_LOANS::VARCHAR)=TRY_TO_NUMBER(b.LOAN_COUNT::VARCHAR)
          AND TRY_TO_DOUBLE(b.NOMINAL_KNOWN_SUM::VARCHAR)>=50000
          AND k.D<=DATEADD('MONTH',-1,DATE '{END}'),1,0)) HEAD_ELIGIBLE
 FROM HEAD_KEYS k JOIN {T} t ON TRIM(TO_VARCHAR(t.ENTITY_ID))=k.HEAD_ID AND TO_DATE(t.MAP_DTE)=k.D
 LEFT JOIN {B} b ON TRIM(TO_VARCHAR(b.ENTITY_ID))=k.HEAD_ID AND TO_DATE(b.MAP_DTE)=k.D
 GROUP BY 1,2
), X AS (
 SELECT s.*,w.M,f.GRADE,f.FEATURE,he.HEAD_ELIGIBLE,
        LAG(s.D) OVER(PARTITION BY s.STAGE,s.MEMBER_ID,s.HEAD_ID,w.M ORDER BY s.D) PREV_D,
        LAG(f.FEATURE) OVER(PARTITION BY s.STAGE,s.MEMBER_ID,s.HEAD_ID,w.M ORDER BY s.D) PREV_F,
        LAG(f.GRADE) OVER(PARTITION BY s.STAGE,s.MEMBER_ID,s.HEAD_ID,w.M ORDER BY s.D) PREV_GRADE
 FROM STAGES s CROSS JOIN W w
 LEFT JOIN FEATURES f ON f.MEMBER_ID=s.MEMBER_ID AND f.D=s.D AND f.M=w.M
 LEFT JOIN HEAD_ELIG he ON he.HEAD_ID=s.HEAD_ID AND he.D=s.D
), FLAGS AS (
 SELECT *,DATEDIFF('DAY',PREV_D,D) GAP_DAYS FROM X
), SUMMARY AS (
 SELECT STAGE,M LOOKBACK_MONTHS,
 COUNT_IF(GRADE IS NOT NULL) N_GRADE_DAYS,
 COUNT(DISTINCT IFF(GRADE IS NOT NULL,MEMBER_ID,NULL)) N_GRADED_MEMBERS,
 COUNT_IF(FEATURE IS NOT NULL) N_FEATURE_DAYS,
 COUNT(DISTINCT IFF(FEATURE IS NOT NULL,MEMBER_ID,NULL)) N_FEATURE_MEMBERS,
 COUNT(DISTINCT IFF(FEATURE IS NOT NULL AND PREV_F IS NOT NULL AND GAP_DAYS BETWEEN 1 AND 7,MEMBER_ID,NULL)) N_COMPARABLE_MEMBERS_7D,
 COUNT_IF(FEATURE IS NOT NULL AND PREV_F IS NOT NULL AND GAP_DAYS BETWEEN 1 AND 7) N_COMPARABLE_DAYS_7D,
 COUNT_IF(FEATURE>=1) N_POSITIVE_FEATURE_DAYS,
 COUNT_IF(GRADE>PREV_GRADE) N_OBSERVED_GRADE_INCREASES,
 COUNT_IF(FEATURE>=1 AND PREV_F<1 AND GAP_DAYS BETWEEN 1 AND 7) N_CROSS_1_7D,
 COUNT_IF(FEATURE>=2 AND PREV_F<2 AND GAP_DAYS BETWEEN 1 AND 7) N_CROSS_2_7D,
 COUNT_IF(FEATURE>=3 AND PREV_F<3 AND GAP_DAYS BETWEEN 1 AND 7) N_CROSS_3_7D,
 COUNT_IF(FEATURE>=1 AND PREV_F<1 AND GAP_DAYS BETWEEN 8 AND 30) N_CROSS_1_8_TO_30D,
 COUNT_IF(FEATURE>=1 AND PREV_F IS NULL) N_POSITIVE_WITHOUT_PREV,
 COUNT(DISTINCT IFF(FEATURE>=1 AND PREV_F<1 AND GAP_DAYS BETWEEN 1 AND 7 AND HEAD_ELIGIBLE=1,HEAD_ID||'|'||TO_VARCHAR(D),NULL)) N_ELIGIBLE_HEAD_DAYS_CROSS_1
 FROM FLAGS GROUP BY 1,2
)
SELECT * FROM SUMMARY
UNION ALL
SELECT 'REASON_'||r.REASON,0,
 COUNT_IF(s.GRADE IS NOT NULL),COUNT(DISTINCT IFF(s.GRADE IS NOT NULL,r.MEMBER_ID,NULL)),
 NULL,NULL,NULL,NULL,NULL,NULL,NULL,NULL,NULL,NULL,NULL,NULL
FROM REASON_ROWS r LEFT JOIN SLOT s ON s.MEMBER_ID=r.MEMBER_ID AND s.D=r.D
GROUP BY r.REASON
ORDER BY STAGE,LOOKBACK_MONTHS
"""
result = SQLExecutor2(connection='DEV_AA_TEAMS').query_to_df(sql)
print(result.to_string(index=False), flush=True)
print('FINISHED | No datasets modified', flush=True)
