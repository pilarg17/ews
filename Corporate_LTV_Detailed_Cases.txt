WITH

    SELECT TRIM(TO_CHAR(SOURCE_CONTRACT_REFERENCE)) INSTRUMENT_ID,
           COUNT(*) N_ROWS,
           COUNT(DISTINCT OUTSTANDING_NOMINAL_AMOUNT) N_NOMINAL_VALUES,
           COUNT(DISTINCT NVL(OFF_BALANCE_SHEET_AMOUNT,0)) N_OFF_VALUES,
           SUM(CASE WHEN OUTSTANDING_NOMINAL_AMOUNT IS NULL OR
                         OUTSTANDING_NOMINAL_AMOUNT < 0 OR
                         OUTSTANDING_NOMINAL_AMOUNT >= 11111111111 OR
                         NVL(OFF_BALANCE_SHEET_AMOUNT,0) < 0
                    THEN 1 ELSE 0 END) N_INVALID_EXPOSURE,
           MAX(OUTSTANDING_NOMINAL_AMOUNT) NOMINAL_REG,
           MAX(NVL(OFF_BALANCE_SHEET_AMOUNT,0)) OFF_BALANCE_REG
    FROM {reg}
    WHERE REFERENCE_DATE={reg_int}
      AND SOURCE_CONTRACT_REFERENCE IS NOT NULL
      AND SOURCE_CONTRACT_REFERENCE NOT LIKE '211MD%'
    GROUP BY TRIM(TO_CHAR(SOURCE_CONTRACT_REFERENCE))
),
REG AS (
    SELECT INSTRUMENT_ID,
           CASE WHEN N_INVALID_EXPOSURE=0 AND N_NOMINAL_VALUES=1 AND
                     N_OFF_VALUES<=1 AND NOMINAL_REG+OFF_BALANCE_REG>0
                THEN NOMINAL_REG+OFF_BALANCE_REG END EXPOSURE,
           N_ROWS,
           CASE WHEN N_INVALID_EXPOSURE=0 AND N_NOMINAL_VALUES=1 AND
                     N_OFF_VALUES<=1 AND NOMINAL_REG+OFF_BALANCE_REG>0
                THEN 1 ELSE 0 END EXPOSURE_VALID
    FROM REG_BASE
),
LINKS AS (
    SELECT TRIM(TO_CHAR(INSTRUMENT_ID)) INSTRUMENT_ID,
           TRIM(TO_CHAR(PROTECTION_ID)) PROTECTION_ID,
           COUNT(*) N_RAW_LINKS
    FROM {links}
    WHERE MAP_DTE >= DATE '{d}' AND MAP_DTE < DATE '{d}'+1
      AND INSTRUMENT_ID IS NOT NULL AND PROTECTION_ID IS NOT NULL
    GROUP BY TRIM(TO_CHAR(INSTRUMENT_ID)),TRIM(TO_CHAR(PROTECTION_ID))
),
PROTECTION_BASE AS (
    SELECT TRIM(TO_CHAR(PROTECTION_ID)) PROTECTION_ID,
           COUNT(DISTINCT TRIM(TO_CHAR(PROTECTION_TYPE))) N_TYPES,
           MAX(TRIM(TO_CHAR(PROTECTION_TYPE))) PROTECTION_TYPE,
           COUNT(*) N_PROTECTION_ROWS,
           CASE WHEN COUNT(DISTINCT TRIM(PROTECTION_INTERNAL_SUBTYPE_DESC))=1
                 AND COUNT(TRIM(PROTECTION_INTERNAL_SUBTYPE_DESC))=COUNT(*)
                THEN MAX(TRIM(PROTECTION_INTERNAL_SUBTYPE_DESC))
           END PROTECTION_INTERNAL_SUBTYPE_DESC
    FROM {prot}
    WHERE MAP_DTE >= DATE '{d}' AND MAP_DTE < DATE '{d}'+1
      AND PROTECTION_ID IS NOT NULL
    GROUP BY TRIM(TO_CHAR(PROTECTION_ID))
),
METRICS_BASE AS (
    SELECT TRIM(TO_CHAR(PROTECTION_ID)) PROTECTION_ID,
           COUNT(*) N_METRIC_ROWS,
           COUNT(DISTINCT NET_MARKET_VALUE_LCY) N_NET_VALUES,
           COUNT(DISTINCT COLLATERAL_COVER_AMOUNT_LCY) N_COVER_VALUES,
           MAX(NET_MARKET_VALUE_LCY) NET_VALUE,
           MAX(COLLATERAL_COVER_AMOUNT_LCY) COVER_VALUE
    FROM {metrics}
    WHERE VALID_DATE >= DATE '{m}' AND VALID_DATE < DATE '{m}'+1
      AND MAP_DTE < DATE '{d}'+1
      AND PROTECTION_ID IS NOT NULL
    GROUP BY TRIM(TO_CHAR(PROTECTION_ID))
),
PROTECTION_VALUES AS (
    SELECT P.PROTECTION_ID, P.PROTECTION_TYPE, P.N_TYPES,
           P.PROTECTION_INTERNAL_SUBTYPE_DESC,
           CASE WHEN P.PROTECTION_TYPE IN ({allowed}) AND P.N_TYPES=1
                     THEN 1 ELSE 0 END ELIGIBLE,
           CASE WHEN M.N_NET_VALUES>1 OR M.N_COVER_VALUES>1 THEN 1 ELSE 0 END AMBIGUOUS_METRICS,
           CASE WHEN M.N_NET_VALUES<=1 AND M.NET_VALUE>2 THEN M.NET_VALUE
                WHEN M.N_COVER_VALUES<=1 AND M.COVER_VALUE>2 THEN M.COVER_VALUE
                ELSE NULL END MARKET_OR_COVER_VALUE
    FROM PROTECTION_BASE P
    LEFT JOIN METRICS_BASE M ON P.PROTECTION_ID=M.PROTECTION_ID
),
SHARED AS (
    SELECT L.PROTECTION_ID,
           COUNT(*) N_LINKED_INSTRUMENTS,
           SUM(CASE WHEN L.N_RAW_LINKS<>1 THEN 1 ELSE 0 END) N_DUP_LINKS,
           SUM(CASE WHEN R.EXPOSURE_VALID=1 THEN R.EXPOSURE ELSE 0 END) SHARED_EXPOSURE,
           SUM(CASE WHEN NVL(R.EXPOSURE_VALID,0)<>1 THEN 1 ELSE 0 END) N_BAD_EXPOSURES
    FROM LINKS L
    LEFT JOIN REG R ON L.INSTRUMENT_ID=R.INSTRUMENT_ID
    GROUP BY L.PROTECTION_ID
),
PAIR AS (
    SELECT L.INSTRUMENT_ID, L.PROTECTION_ID, L.N_RAW_LINKS,
           P.PROTECTION_INTERNAL_SUBTYPE_DESC,
           NVL(P.ELIGIBLE,0) ELIGIBLE,
           CASE WHEN P.PROTECTION_ID IS NULL OR P.N_TYPES<>1 THEN 1 ELSE 0 END UNKNOWN_PROTECTION_TYPE,
           NVL(P.AMBIGUOUS_METRICS,0) AMBIGUOUS_METRICS,
           P.MARKET_OR_COVER_VALUE COLLATERAL_VALUE,
           S.N_LINKED_INSTRUMENTS, S.N_DUP_LINKS,
           S.N_BAD_EXPOSURES, S.SHARED_EXPOSURE,
           R.EXPOSURE,
           CASE WHEN P.ELIGIBLE=1 AND L.N_RAW_LINKS=1
                     AND S.N_DUP_LINKS=0 AND S.N_BAD_EXPOSURES=0
                     AND S.SHARED_EXPOSURE>0 AND P.AMBIGUOUS_METRICS=0
                     AND P.MARKET_OR_COVER_VALUE>2 AND R.EXPOSURE_VALID=1
                THEN P.MARKET_OR_COVER_VALUE * R.EXPOSURE / S.SHARED_EXPOSURE
                ELSE NULL END ALLOCATED_COLLATERAL_VALUE
    FROM LINKS L
    LEFT JOIN PROTECTION_VALUES P ON L.PROTECTION_ID=P.PROTECTION_ID
    JOIN SHARED S ON L.PROTECTION_ID=S.PROTECTION_ID
    LEFT JOIN REG R ON L.INSTRUMENT_ID=R.INSTRUMENT_ID
),
AGG AS (
    SELECT INSTRUMENT_ID,
           COUNT(*) N_TOTAL_PROTECTION_LINKS,
           COUNT(DISTINCT CASE WHEN ELIGIBLE=1 THEN PROTECTION_INTERNAL_SUBTYPE_DESC END)
                N_DISTINCT_PROTECTION_SUBTYPES,
           SUM(CASE WHEN ELIGIBLE=1 AND PROTECTION_INTERNAL_SUBTYPE_DESC IS NULL THEN 1 ELSE 0 END)
                N_MISSING_PROTECTION_SUBTYPES,
           SUM(CASE WHEN ELIGIBLE=1 THEN 1 ELSE 0 END) N_REAL_ESTATE_LINKS,
           SUM(CASE WHEN UNKNOWN_PROTECTION_TYPE=1 THEN 1 ELSE 0 END) N_UNCLASSIFIED_LINKS,
           SUM(CASE WHEN ELIGIBLE=1 AND N_LINKED_INSTRUMENTS>1 THEN 1 ELSE 0 END) N_SHARED_REAL_ESTATE_LINKS,
           SUM(CASE WHEN ELIGIBLE=1 AND ALLOCATED_COLLATERAL_VALUE IS NULL THEN 1 ELSE 0 END) N_INCOMPLETE_REAL_ESTATE_LINKS,
           SUM(CASE WHEN ELIGIBLE=1 AND AMBIGUOUS_METRICS=1 THEN 1 ELSE 0 END) N_AMBIGUOUS_VALUATIONS,
           SUM(CASE WHEN ELIGIBLE=1 AND N_BAD_EXPOSURES>0 THEN 1 ELSE 0 END) N_BAD_SHARED_EXPOSURES,
           SUM(CASE WHEN ELIGIBLE=1 AND N_DUP_LINKS>0 THEN 1 ELSE 0 END) N_DUPLICATE_LINKS,
           SUM(CASE WHEN ELIGIBLE=1 THEN ALLOCATED_COLLATERAL_VALUE END) ALLOCATED_COLLATERAL_VALUE,
           SUM(CASE WHEN ELIGIBLE=1 AND ALLOCATED_COLLATERAL_VALUE>0 THEN 1 ELSE 0 END) N_VALID_REAL_ESTATE_LINKS
    FROM PAIR GROUP BY INSTRUMENT_ID
),
SUBTYPES_DISTINCT AS (
    SELECT DISTINCT INSTRUMENT_ID, PROTECTION_INTERNAL_SUBTYPE_DESC
    FROM PAIR WHERE ELIGIBLE=1 AND PROTECTION_INTERNAL_SUBTYPE_DESC IS NOT NULL
),
SUBTYPES_AGG AS (
    SELECT INSTRUMENT_ID,
           LISTAGG(PROTECTION_INTERNAL_SUBTYPE_DESC, ' | ')
             WITHIN GROUP (ORDER BY PROTECTION_INTERNAL_SUBTYPE_DESC) PROTECTION_INTERNAL_SUBTYPE_DESC
    FROM SUBTYPES_DISTINCT GROUP BY INSTRUMENT_ID
)
SELECT 'I|'||A.INSTRUMENT_ID INSTRUMENT_ID,
       R.EXPOSURE AS REGULATORY_EXPOSURE,
       A.N_TOTAL_PROTECTION_LINKS, A.N_REAL_ESTATE_LINKS, A.N_UNCLASSIFIED_LINKS,
       A.N_SHARED_REAL_ESTATE_LINKS, A.N_INCOMPLETE_REAL_ESTATE_LINKS,
       A.N_AMBIGUOUS_VALUATIONS, A.N_BAD_SHARED_EXPOSURES,
       A.N_DUPLICATE_LINKS, A.N_VALID_REAL_ESTATE_LINKS,
       CASE WHEN A.N_REAL_ESTATE_LINKS>0 AND A.N_INCOMPLETE_REAL_ESTATE_LINKS=0
                AND A.N_VALID_REAL_ESTATE_LINKS=A.N_REAL_ESTATE_LINKS
                AND A.ALLOCATED_COLLATERAL_VALUE>0
                THEN A.ALLOCATED_COLLATERAL_VALUE END ALLOCATED_COLLATERAL_VALUE,
       CASE WHEN R.EXPOSURE_VALID=1 AND A.N_REAL_ESTATE_LINKS>0
                AND A.N_INCOMPLETE_REAL_ESTATE_LINKS=0
                AND A.N_VALID_REAL_ESTATE_LINKS=A.N_REAL_ESTATE_LINKS
                AND A.ALLOCATED_COLLATERAL_VALUE>0
                THEN R.EXPOSURE/A.ALLOCATED_COLLATERAL_VALUE END LTV_RECONSTRUCTED,
       S.PROTECTION_INTERNAL_SUBTYPE_DESC,
       A.N_DISTINCT_PROTECTION_SUBTYPES,
       A.N_MISSING_PROTECTION_SUBTYPES
FROM AGG A
LEFT JOIN REG R ON A.INSTRUMENT_ID=R.INSTRUMENT_ID
LEFT JOIN SUBTYPES_AGG S ON A.INSTRUMENT_ID=S.INSTRUMENT_ID
""".format(reg=ORACLE_REG, reg_int=reg_int, links=ORACLE_LINK,
           prot=ORACLE_PROTECTION,metrics=ORACLE_METRICS,d=reg_d,m=metrics_d,allowed=allowed)

def read_all_pairs(reg_day,metric_day,ids):
    key=reg_day.replace('-','')
    id_list=','.join(sf_literal(x) for x in ids)
    q=retail_ltv_sql(reg_day,key,metric_day).split(',\nAGG AS (')[0]
    q+=f"""
    , FOCUS_PROTECTIONS AS (SELECT DISTINCT PROTECTION_ID FROM LINKS WHERE INSTRUMENT_ID IN ({id_list}))
    SELECT 'I|'||p.INSTRUMENT_ID INSTRUMENT_ID,
           'P|'||p.PROTECTION_ID PROTECTION_ID,
           CASE WHEN p.INSTRUMENT_ID IN ({id_list}) THEN 1 ELSE 0 END IS_FOCUS,
           p.N_RAW_LINKS,p.ELIGIBLE,p.UNKNOWN_PROTECTION_TYPE,p.AMBIGUOUS_METRICS,
           p.N_LINKED_INSTRUMENTS,p.N_DUP_LINKS,p.N_BAD_EXPOSURES,
           p.SHARED_EXPOSURE,p.EXPOSURE,
           r.NOMINAL_REG,r.OFF_BALANCE_REG,r.N_NOMINAL_VALUES,r.N_OFF_VALUES,r.N_INVALID_EXPOSURE,
           pb.PROTECTION_TYPE,pb.N_TYPES,pb.N_PROTECTION_ROWS,
           pb.PROTECTION_INTERNAL_SUBTYPE_DESC,
           m.N_METRIC_ROWS,m.N_NET_VALUES,m.N_COVER_VALUES,
           m.NET_VALUE NET_MARKET_VALUE_LCY,m.COVER_VALUE COLLATERAL_COVER_AMOUNT_LCY,
           p.COLLATERAL_VALUE RAW_VALUE,p.ALLOCATED_COLLATERAL_VALUE ALLOCATED,
           CASE WHEN m.N_NET_VALUES<=1 AND m.NET_VALUE>2 THEN 'MARKET_VALUE'
                WHEN m.N_COVER_VALUES<=1 AND m.COVER_VALUE>2 THEN 'COVER_AMOUNT'
                ELSE 'MISSING' END VALUE_SOURCE
    FROM PAIR p JOIN FOCUS_PROTECTIONS f ON p.PROTECTION_ID=f.PROTECTION_ID
    LEFT JOIN PROTECTION_BASE pb ON p.PROTECTION_ID=pb.PROTECTION_ID
    LEFT JOIN METRICS_BASE m ON p.PROTECTION_ID=m.PROTECTION_ID
    LEFT JOIN REG_BASE r ON p.INSTRUMENT_ID=r.INSTRUMENT_ID
    """
    print('READING | unfiltered original Oracle CTEs |',reg_day,metric_day,flush=True)
    x=query(ora,q)
    for col in ['INSTRUMENT_ID','PROTECTION_ID']: x[col]=x[col].astype(str).str[2:]
    assert not x.duplicated(['INSTRUMENT_ID','PROTECTION_ID']).any(),'Non-unique pair keys'
    x['REG_DATE']=reg_day;x['METRICS_DATE']=metric_day
    return x

# ============================================================
# 4. Decompose changes only where both stored snapshots reconcile.
# ============================================================
cols=['INSTRUMENT_ID','PROTECTION_ID','PAIR_STATUS','ELIGIBLE','RAW_VALUE','VALUE_SOURCE','WEIGHT','ALLOCATED','NET_MARKET_VALUE_LCY','COLLATERAL_COVER_AMOUNT_LCY','N_LINKED_INSTRUMENTS']
a=focus.loc[focus.SIDE.eq('OLD'),cols]; b=focus.loc[focus.SIDE.eq('NEW'),cols]
ltv_pair_comparison=a.merge(b,on=['INSTRUMENT_ID','PROTECTION_ID'],how='outer',suffixes=('_OLD','_NEW'),indicator=True,validate='one_to_one')
c=ltv_pair_comparison
old_valid=c.PAIR_STATUS_OLD.eq('VALID');new_valid=c.PAIR_STATUS_NEW.eq('VALID');retained=old_valid&new_valid
# A valuation source change is a distinct category even if market value also changes.
c['PAIR_DRIVER']=np.select([c._merge.eq('left_only'),c._merge.eq('right_only'),c.ELIGIBLE_OLD.ne(c.ELIGIBLE_NEW),~retained,c.VALUE_SOURCE_OLD.ne(c.VALUE_SOURCE_NEW),c.RAW_VALUE_NEW.lt(c.RAW_VALUE_OLD-.01)&c.VALUE_SOURCE_NEW.eq('MARKET_VALUE'),c.RAW_VALUE_NEW.lt(c.RAW_VALUE_OLD-.01),c.RAW_VALUE_NEW.gt(c.RAW_VALUE_OLD+.01),c.WEIGHT_NEW.sub(c.WEIGHT_OLD).abs().gt(1e-10)],['RAW_LINK_ABSENT_NEW_SNAPSHOT','RAW_LINK_ABSENT_OLD_SNAPSHOT','ELIGIBILITY_CHANGED','PAIR_NOT_COMPARABLE','VALUATION_FIELD_CHANGED','MARKET_VALUE_DECREASE','COVER_AMOUNT_DECREASE','VALUATION_INCREASE','ALLOCATION_WEIGHT_CHANGED'],default='UNCHANGED')
c['VALUE_EFFECT']=((c.RAW_VALUE_NEW-c.RAW_VALUE_OLD)*c.WEIGHT_OLD).where(retained,0)
c['WEIGHT_EFFECT']=(c.RAW_VALUE_NEW*(c.WEIGHT_NEW-c.WEIGHT_OLD)).where(retained,0)
c['LINK_OR_ELIGIBILITY_EFFECT']=np.select([old_valid&~new_valid,~old_valid&new_valid],[-c.ALLOCATED_OLD,c.ALLOCATED_NEW],default=0)
c=c.merge(peak[['ENTITY_ID','INSTRUMENT_ID']],on='INSTRUMENT_ID',how='left',validate='many_to_one')
ltv_pair_comparison=c
fx=c.groupby('INSTRUMENT_ID')[['VALUE_EFFECT','WEIGHT_EFFECT','LINK_OR_ELIGIBILITY_EFFECT']].sum(min_count=1).reset_index()
statuses=r.pivot(index='INSTRUMENT_ID',columns='SIDE',values='RECON_STATUS').rename(columns={'OLD':'RECON_OLD','NEW':'RECON_NEW'}).reset_index()
check=peak.merge(fx,on='INSTRUMENT_ID',how='left',validate='one_to_one').merge(statuses,on='INSTRUMENT_ID',validate='one_to_one')
exp=stored.pivot(index='INSTRUMENT_ID',columns='SIDE',values='REGULATORY_EXPOSURE').rename(columns={'OLD':'EXPOSURE_OLD','NEW':'EXPOSURE_NEW'}).reset_index()
check=check.merge(exp,on='INSTRUMENT_ID',validate='one_to_one')
check['EXPOSURE_CHANGE']=check.EXPOSURE_NEW-check.EXPOSURE_OLD
check['COLLATERAL_CHANGE']=check.ALLOCATED_NEW-check.ALLOCATED_OLD
check['RESIDUAL']=check.COLLATERAL_CHANGE-check.VALUE_EFFECT-check.WEIGHT_EFFECT-check.LINK_OR_ELIGIBILITY_EFFECT
check['DECOMPOSITION_OK']=check.RECON_OLD.eq('MATCH')&check.RECON_NEW.eq('MATCH')&check.RESIDUAL.abs().le(.1)
# Do not label partially matched pair changes as confirmed customer-level causes.
c=c.merge(check[['INSTRUMENT_ID','DECOMPOSITION_OK']],on='INSTRUMENT_ID',validate='many_to_one');ltv_pair_comparison=c
print('\n=== FULL LOAN DECOMPOSITION | ALL IDS ===')
print(check.sort_values(['ENTITY_ID','INSTRUMENT_ID']).round(4).to_string(index=False))
print('\nFully reconciled loans:',int(check.DECOMPOSITION_OK.sum()),'/ 30')
print('\n=== PAIR CHANGES | ALL IDS AND COMPARABILITY ===')
print(c.sort_values(['ENTITY_ID','INSTRUMENT_ID','PROTECTION_ID']).round(4).to_string(index=False))
missing_collateral_cases=check.loc[~check.DECOMPOSITION_OK].copy()
print('\n=== UNRECONCILED LOAN IDS ===')
print(missing_collateral_cases[['ENTITY_ID','INSTRUMENT_ID','RECON_OLD','RECON_NEW','ALLOCATED_OLD','ALLOCATED_NEW']].to_string(index=False))

 AND m.POLICY_THRESHOLD>0 AND p.LTV_VALUE<=p.POLICY_THRESHOLD AND m.LTV_VALUE>m.POLICY_THRESHOLD
)
SELECT 'E|'||c.ENTITY_ID ENTITY_ID,'I|'||c.INSTRUMENT_ID INSTRUMENT_ID,c.* EXCLUDE(ENTITY_ID,INSTRUMENT_ID),
 s.TARGET,s.EVENT_NOMINAL,t.TGT_DEFAULT_SAMEDAY,t.TGT_FORBEARANCE_SAMEDAY,t.TGT_DIM1_SAMEDAY,t.TGT_ANY_SAMEDAY,
 b.PD_CRR,b.DPD_MAX,t.ACA_FINAL_RATING_WORST SLOTTING_GRADE,t.ACA_SLOT_GRADE_CONFLICT GRADE_CONFLICT,t.ACA_SLOT_ADC_ELIGIBLE,
 CASE WHEN s.ENTITY_ID IS NULL THEN 'OUTSIDE_SCOPE' WHEN s.TARGET IS NULL THEN 'TARGET_UNKNOWN' WHEN s.TARGET<>0 THEN 'TARGET_ALREADY_ACTIVE'
 WHEN s.EVENT_NOMINAL IS NULL THEN 'EXPOSURE_INCOMPLETE' WHEN s.EVENT_NOMINAL<50000 THEN 'EXPOSURE_BELOW_50K'
 WHEN c.SIGNAL_DATE>DATEADD('MONTH',-1,DATE '{END}') THEN 'RIGHT_CENSORED' ELSE 'ELIGIBLE' END EVALUATION_STATUS
FROM ADC_CROSS c LEFT JOIN SCOPE s ON s.PORTFOLIO='ADC' AND s.ENTITY_ID=c.ENTITY_ID AND s.MAP_DTE=c.SIGNAL_DATE
LEFT JOIN {T} t ON TRIM(TO_VARCHAR(t.ENTITY_ID))=c.ENTITY_ID AND TO_DATE(t.MAP_DTE)=c.SIGNAL_DATE
LEFT JOIN {B} b ON TRIM(TO_VARCHAR(b.ENTITY_ID))=c.ENTITY_ID AND TO_DATE(b.MAP_DTE)=c.SIGNAL_DATE
"""

# ============================================================
# 6. ACA original records at each case date; exact and prior seven days.
#    Raw source evidence only: do not infer a target from a non-default grade.
# ============================================================
keys=missing_target_cases[['ENTITY_ID','SIGNAL_DATE']].drop_duplicates().copy()
keys.SIGNAL_DATE=pd.to_datetime(keys.SIGNAL_DATE,utc=True).dt.strftime('%Y-%m-%d')
keys_sql=' UNION ALL '.join(f"SELECT {sf_literal(x.ENTITY_ID)} ENTITY_ID,DATE '{x.SIGNAL_DATE}' SIGNAL_DATE FROM DUAL" for x in keys.itertuples(index=False))
aca_sql=f"""
WITH FOCUS AS ({keys_sql}),
RAW_ACA AS (
 SELECT f.ENTITY_ID,f.SIGNAL_DATE,TRUNC(a.MAP_DTE) ACA_SNAPSHOT,
 TRIM(TO_CHAR(a.PROJECT_NUMBER)) PROJECT_NUMBER,TRIM(TO_CHAR(a.RISK_ASSESSMENT_ID)) ASSESSMENT_ID,
 TRIM(TO_CHAR(a.MODEL_ID)) MODEL_ID,TRUNC(a.RISKASSESSMENT_DATE) REVIEW_DATE,
 TRUNC(a.RISKASSESSMENT_APPROVAL_DATE) APPROVAL_DATE,
 TRIM(TO_CHAR(a.RISKASSESSMENT_STATUS)) ASSESSMENT_STATUS,
 TRIM(TO_CHAR(a.FINAL_RATING)) FINAL_RATING
 FROM FOCUS f JOIN BDWH_PLY.PLY_ACA_LEGAL_ENTITY_DAY_VW a
 ON TRIM(TO_CHAR(a.LEGAL_ENTITY_REF_CODE))=f.ENTITY_ID
 AND a.MAP_DTE>=f.SIGNAL_DATE-7 AND a.MAP_DTE<f.SIGNAL_DATE+1
), LATEST AS (SELECT ENTITY_ID,SIGNAL_DATE,MAX(ACA_SNAPSHOT) LATEST_ACA_SNAPSHOT FROM RAW_ACA GROUP BY ENTITY_ID,SIGNAL_DATE)
SELECT DISTINCT 'E|'||f.ENTITY_ID ENTITY_ID,f.SIGNAL_DATE,l.LATEST_ACA_SNAPSHOT,
 CASE WHEN r.ACA_SNAPSHOT=f.SIGNAL_DATE THEN 'EXACT' WHEN r.ACA_SNAPSHOT IS NOT NULL THEN 'PRIOR_7D' ELSE 'NO_RECORD_WITHIN_7D' END SOURCE_STATUS,
 r.PROJECT_NUMBER,r.ASSESSMENT_ID,r.MODEL_ID,r.REVIEW_DATE,r.APPROVAL_DATE,r.ASSESSMENT_STATUS,r.FINAL_RATING,
 CASE WHEN r.ASSESSMENT_STATUS='3' AND r.APPROVAL_DATE<=f.SIGNAL_DATE AND r.REVIEW_DATE<=f.SIGNAL_DATE AND r.FINAL_RATING IN ('1','2','3','4','5') THEN 1 ELSE 0 END APPROVED_GRADE_CANDIDATE
FROM FOCUS f LEFT JOIN LATEST l ON l.ENTITY_ID=f.ENTITY_ID AND l.SIGNAL_DATE=f.SIGNAL_DATE
LEFT JOIN RAW_ACA r ON r.ENTITY_ID=f.ENTITY_ID AND r.SIGNAL_DATE=f.SIGNAL_DATE AND r.ACA_SNAPSHOT=l.LATEST_ACA_SNAPSHOT
"""
aca_missing_cases=query(ora,aca_sql);aca_missing_cases.ENTITY_ID=aca_missing_cases.ENTITY_ID.astype(str).str[2:]
