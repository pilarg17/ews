      ORDER BY SIGNAL_DATE ROWS UNBOUNDED PRECEDING) SIGNAL_GROUP
 FROM SIGNAL_PREVIOUS s
),
SIGNALS AS (
 SELECT PORTFOLIO,ENTITY_ID,MIN(SIGNAL_DATE) SIGNAL_DATE,
        INDICATOR,THRESHOLD_A,THRESHOLD_B
 FROM SIGNAL_GROUPS
 GROUP BY PORTFOLIO,ENTITY_ID,INDICATOR,THRESHOLD_A,THRESHOLD_B,SIGNAL_GROUP
),
MATCHES AS (
 SELECT s.*,e.EVENT_ID,e.EVENT_DATE,e.EVENT_NOMINAL,
        DATEDIFF('DAY',s.SIGNAL_DATE,e.EVENT_DATE)/(365.25/12.0) LEAD_MONTHS
 FROM SIGNALS s JOIN EVENTS e ON s.PORTFOLIO=e.PORTFOLIO AND s.ENTITY_ID=e.ENTITY_ID
 WHERE s.SIGNAL_DATE BETWEEN DATEADD('MONTH',-11,e.EVENT_DATE)
                        AND DATEADD('MONTH',-1,e.EVENT_DATE)
),
TP_FIRST AS (
 SELECT m.* FROM MATCHES m
 QUALIFY ROW_NUMBER() OVER (PARTITION BY PORTFOLIO,INDICATOR,THRESHOLD_A,THRESHOLD_B,EVENT_ID
 ORDER BY SIGNAL_DATE)=1
),
TP AS (
 SELECT PORTFOLIO,INDICATOR,THRESHOLD_A,THRESHOLD_B,
        COUNT(*) N_TP,SUM(EVENT_NOMINAL) SUM_NOMINAL_TP,
        SUM(EVENT_NOMINAL*(0.40*(1.0-0.50*(LEAST(11,GREATEST(1,LEAD_MONTHS))-1)/10.0))) SUM_TP_COST
 FROM TP_FIRST GROUP BY 1,2,3,4
),
SIGNAL_MATCHED AS (
 SELECT DISTINCT PORTFOLIO,ENTITY_ID,SIGNAL_DATE,INDICATOR,THRESHOLD_A,THRESHOLD_B
 FROM MATCHES
),
SIGNAL_CLASS AS (
 SELECT s.PORTFOLIO,s.INDICATOR,s.THRESHOLD_A,s.THRESHOLD_B,
        COUNT(*) N_EVALUATED_EPISODES,
        COUNT_IF(m.ENTITY_ID IS NOT NULL) N_TP_ALERTS,
        COUNT_IF(m.ENTITY_ID IS NULL) N_FP
 FROM SIGNALS s LEFT JOIN SIGNAL_MATCHED m
 ON s.PORTFOLIO=m.PORTFOLIO AND s.ENTITY_ID=m.ENTITY_ID
 AND s.SIGNAL_DATE=m.SIGNAL_DATE AND s.INDICATOR=m.INDICATOR
 AND s.THRESHOLD_A=m.THRESHOLD_A AND s.THRESHOLD_B=m.THRESHOLD_B
 GROUP BY 1,2,3,4
),
EVENT_TOTAL AS (
 SELECT PORTFOLIO,COUNT(*) N_EVENTS,SUM(EVENT_NOMINAL) SUM_EVENT_NOMINAL
 FROM EVENTS GROUP BY 1
),
EVENT_DETECTABLE AS (
 SELECT e.PORTFOLIO,s.INDICATOR,s.THRESHOLD_A,s.THRESHOLD_B,
        COUNT(DISTINCT e.EVENT_ID) N_EVENTS_DETECTABLE
 FROM EVENTS e JOIN SOURCE_AVAIL s ON e.PORTFOLIO=s.PORTFOLIO AND e.ENTITY_ID=s.ENTITY_ID
 JOIN SCOPE ep ON ep.PORTFOLIO=s.PORTFOLIO AND ep.ENTITY_ID=s.ENTITY_ID
   AND ep.MAP_DTE=s.SIGNAL_DATE AND ep.TARGET=0 AND ep.EVENT_NOMINAL>=50000
 AND s.SIGNAL_DATE BETWEEN DATEADD('MONTH',-11,e.EVENT_DATE)
                        AND DATEADD('MONTH',-1,e.EVENT_DATE)
 GROUP BY 1,2,3,4
),
ALERT_DAY AS (
 SELECT PORTFOLIO,INDICATOR,THRESHOLD_A,THRESHOLD_B,SIGNAL_DATE,
        COUNT(DISTINCT ENTITY_ID) N_DAILY_ALERTS
 FROM SIGNALS GROUP BY 1,2,3,4,5
),
ALERT_MAX AS (
 SELECT PORTFOLIO,INDICATOR,THRESHOLD_A,THRESHOLD_B,
        MAX(N_DAILY_ALERTS) MAX_DAILY_ALERTS
 FROM ALERT_DAY GROUP BY 1,2,3,4
),
EVAL_DAYS AS (
 SELECT PORTFOLIO,COUNT(DISTINCT MAP_DTE) N_EVALUATION_DAYS
 FROM SCOPE WHERE MAP_DTE<=DATEADD('MONTH',-1,DATE '{end}') GROUP BY 1
),
GRID AS (
 SELECT p.PORTFOLIO,g.INDICATOR,g.THRESHOLD_A,g.THRESHOLD_B FROM PORTS p
 CROSS JOIN GRID_RULES g
),
METRIC_BASE AS (
 SELECT g.PORTFOLIO,g.INDICATOR,g.THRESHOLD_A,g.THRESHOLD_B,
        COALESCE(et.N_EVENTS,0) N_EVENTS,COALESCE(tp.N_TP,0) N_TP,
        COALESCE(sc.N_FP,0) N_FP,COALESCE(sc.N_TP_ALERTS,0) N_TP_ALERTS,
        COALESCE(sc.N_EVALUATED_EPISODES,0) N_EVALUATED_EPISODES,
        COALESCE(ev.N_EVENTS_DETECTABLE,0) N_EVENTS_DETECTABLE,
        COALESCE(q.N_RAW_EPISODES,0) N_RAW_EPISODES,
        COALESCE(q.N_RAW_ENTITIES,0) N_RAW_ENTITIES,
        COALESCE(d.N_EVALUATION_DAYS,0) N_EVALUATION_DAYS,
        COALESCE(am.MAX_DAILY_ALERTS,0) MAX_DAILY_ALERTS,
        et.SUM_EVENT_NOMINAL,tp.SUM_NOMINAL_TP,tp.SUM_TP_COST
 FROM GRID g
 LEFT JOIN EVENT_TOTAL et ON g.PORTFOLIO=et.PORTFOLIO
 LEFT JOIN EVENT_DETECTABLE ev ON g.PORTFOLIO=ev.PORTFOLIO AND g.INDICATOR=ev.INDICATOR
  AND g.THRESHOLD_A=ev.THRESHOLD_A AND g.THRESHOLD_B=ev.THRESHOLD_B
 LEFT JOIN TP tp ON g.PORTFOLIO=tp.PORTFOLIO AND g.INDICATOR=tp.INDICATOR
  AND g.THRESHOLD_A=tp.THRESHOLD_A AND g.THRESHOLD_B=tp.THRESHOLD_B
 LEFT JOIN SIGNAL_CLASS sc ON g.PORTFOLIO=sc.PORTFOLIO AND g.INDICATOR=sc.INDICATOR
  AND g.THRESHOLD_A=sc.THRESHOLD_A AND g.THRESHOLD_B=sc.THRESHOLD_B
 LEFT JOIN RAW_QA q ON g.PORTFOLIO=q.PORTFOLIO AND g.INDICATOR=q.INDICATOR
  AND g.THRESHOLD_A=q.THRESHOLD_A AND g.THRESHOLD_B=q.THRESHOLD_B
 LEFT JOIN EVAL_DAYS d ON g.PORTFOLIO=d.PORTFOLIO
 LEFT JOIN ALERT_MAX am ON g.PORTFOLIO=am.PORTFOLIO AND g.INDICATOR=am.INDICATOR
  AND g.THRESHOLD_A=am.THRESHOLD_A AND g.THRESHOLD_B=am.THRESHOLD_B
)
SELECT m.PORTFOLIO,m.INDICATOR,'FULL_EWS' TARGET_SCOPE,
       m.THRESHOLD_A,m.THRESHOLD_B,NULL::DOUBLE THRESHOLD_C,
       m.N_EVENTS,m.N_TP,m.N_FP,GREATEST(m.N_EVENTS-m.N_TP,0) N_FN,
       m.N_TP+m.N_FP N_ALERTS,m.N_EVALUATED_EPISODES,m.N_TP_ALERTS,
       IFF(m.N_TP+m.N_FP>0,ROUND(m.N_TP/(m.N_TP+m.N_FP),4),NULL) ACCURACY,
       IFF(m.N_EVENTS>0,ROUND(m.N_TP/m.N_EVENTS,4),NULL) COVERAGE,
       IFF(m.N_EVENTS_DETECTABLE>0,ROUND(m.N_TP/m.N_EVENTS_DETECTABLE,4),NULL) COVERAGE_2,
       IFF(MAX(m.N_TP) OVER (PARTITION BY m.PORTFOLIO,m.INDICATOR,m.THRESHOLD_B)>0,
           ROUND(m.N_TP/MAX(m.N_TP) OVER(PARTITION BY m.PORTFOLIO,m.INDICATOR,m.THRESHOLD_B),4),NULL) COVERAGE_3,
       IFF(m.N_EVALUATED_EPISODES>0,ROUND(m.N_TP_ALERTS/m.N_EVALUATED_EPISODES,4),NULL) ALERT_PRECISION,
       m.N_EVENTS_DETECTABLE,m.N_RAW_EPISODES,m.N_RAW_ENTITIES,
       m.N_EVALUATION_DAYS,m.MAX_DAILY_ALERTS,
       IFF(m.N_EVALUATION_DAYS>0,ROUND(m.N_EVALUATED_EPISODES/m.N_EVALUATION_DAYS,4),NULL) AVG_DAILY_ALERTS,
       IFF(m.N_EVALUATION_DAYS>0,ROUND(m.N_RAW_EPISODES/m.N_EVALUATION_DAYS,4),NULL) AVG_DAILY_RAW_EPISODES,
       ROUND(500*m.N_FP+COALESCE(m.SUM_TP_COST,0)+0.40*GREATEST(COALESCE(m.SUM_EVENT_NOMINAL,0)-COALESCE(m.SUM_NOMINAL_TP,0),0),2) TOTAL_COST_CF1,
       ROUND(1000*m.N_TP+200*m.N_FP+0.5*COALESCE(m.SUM_TP_COST,0)+0.20*GREATEST(COALESCE(m.SUM_EVENT_NOMINAL,0)-COALESCE(m.SUM_NOMINAL_TP,0),0),2) TOTAL_COST,
       'ELIGIBLE_KNOWN_EXPOSURE_ONLY' COST_CF1_STATUS,
       CASE WHEN m.INDICATOR='ltv_policy_breach' AND m.PORTFOLIO NOT IN ('GLOBAL','ADC') THEN 'NOT_ASSESSED_ADC_POLICY_ONLY'
            WHEN m.INDICATOR='ltv_policy_breach' THEN 'ADC_POLICY_MAPPING_PARTIAL_COVERAGE'

            ELSE 'LTV_INCREASE_BACKTEST' END RESULT_STATUS,
       'MONTHLY' DATA_FREQUENCY
FROM METRIC_BASE m ORDER BY m.PORTFOLIO,m.INDICATOR,m.THRESHOLD_A,m.THRESHOLD_B
"""
