SQL = r"""
WITH LINK_RAW AS (
 SELECT TO_DATE(MAP_DTE) MAP_DTE,TRIM(TO_VARCHAR(ENTITY_ID)) MEMBER_ID,
        MIN(TRIM(TO_VARCHAR(GEC_ID))) GEC_ID,
        COUNT(DISTINCT TRIM(TO_VARCHAR(GEC_ID))) N_GROUPS
 FROM {G} WHERE ENTITY_ID IS NOT NULL AND GEC_ID IS NOT NULL
 GROUP BY 1,2
),
LINK_OK AS (SELECT MAP_DTE,MEMBER_ID,GEC_ID FROM LINK_RAW WHERE N_GROUPS=1),
LINK_COUNTS AS (SELECT MAP_DTE,GEC_ID,COUNT(DISTINCT MEMBER_ID) N_LINKED FROM LINK_OK GROUP BY 1,2),
GROUP_HEAD AS (
 SELECT l.MAP_DTE,l.GEC_ID,
        MIN(TRIM(TO_VARCHAR(t.GEC_HEAD_ID))) HEAD_ID,
        COUNT(DISTINCT TRIM(TO_VARCHAR(t.GEC_HEAD_ID))) N_HEADS,
        COUNT_IF(t.GEC_HEAD_ID IS NULL) N_MISSING_HEAD
 FROM LINK_OK l JOIN {T} t ON l.MEMBER_ID=TRIM(TO_VARCHAR(t.ENTITY_ID))
     AND l.MAP_DTE=TO_DATE(t.MAP_DTE)
 JOIN LINK_COUNTS c ON c.GEC_ID=l.GEC_ID AND c.MAP_DTE=l.MAP_DTE
 GROUP BY 1,2
 HAVING N_HEADS=1 AND N_MISSING_HEAD=0
   AND COUNT(DISTINCT l.MEMBER_ID)=MAX(c.N_LINKED)
),
HEAD_VERIFIED AS (
 SELECT g.MAP_DTE,g.GEC_ID,g.HEAD_ID
 FROM GROUP_HEAD g
 JOIN {T} ht ON TRIM(TO_VARCHAR(ht.ENTITY_ID))=g.HEAD_ID
    AND TO_DATE(ht.MAP_DTE)=g.MAP_DTE
 WHERE g.HEAD_ID IS NOT NULL
 GROUP BY 1,2,3
 HAVING COUNT(DISTINCT TRIM(TO_VARCHAR(ht.ENTITY_ID)))=1
),
-- The worst observed project grade is already aggregated at entity-day level.
-- Multiple project grades are not an exclusion: use the worst grade if 1-4.
-- A worst grade of 5 remains excluded, as the non-default project grades
-- cannot be reconstructed from this aggregated input. Never impute grade 4.
-- Keep duplicate entity-day protection separate from project-grade dispersion.
SLOT_SOURCE_RAW AS (
 SELECT TRIM(TO_VARCHAR(ENTITY_ID)) MEMBER_ID,TO_DATE(MAP_DTE) MAP_DTE,
        COUNT(*) N_ROWS,
        MAX(TRY_TO_DOUBLE(TO_VARCHAR(ACA_FINAL_RATING_WORST))) WORST_GRADE,
        MAX(IFF(TRY_TO_BOOLEAN(TO_VARCHAR(ACA_SLOT_GRADE_CONFLICT))=FALSE,1,0)) NO_CONFLICT,
        MAX(IFF(TRY_TO_BOOLEAN(TO_VARCHAR(ACA_SLOT_GRADE_CONFLICT))=TRUE,1,0)) GRADE_CONFLICT
 FROM {T} WHERE ACA_FINAL_RATING_WORST IS NOT NULL
   AND ENTITY_ID IS NOT NULL AND MAP_DTE IS NOT NULL
   AND TO_DATE(MAP_DTE)<=DATE '{end}'
 GROUP BY 1,2
),
SLOT_SOURCE AS (
 SELECT MEMBER_ID,MAP_DTE,
        CASE WHEN N_ROWS=1 AND WORST_GRADE IN (1,2,3,4) THEN WORST_GRADE END GRADE,
        CASE WHEN N_ROWS=1 AND NO_CONFLICT=1 AND WORST_GRADE IN (1,2,3,4)
             THEN WORST_GRADE END PREVIOUS_ELIGIBLE_GRADE,
        WORST_GRADE,GRADE_CONFLICT,N_ROWS
 FROM SLOT_SOURCE_RAW
),
SLOT_HISTORY AS (
 SELECT s.*,
        MAX(IFF(GRADE=1,MAP_DTE,NULL)) OVER(PARTITION BY MEMBER_ID ORDER BY MAP_DTE ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) LAST_GRADE_1,
        MAX(IFF(GRADE=2,MAP_DTE,NULL)) OVER(PARTITION BY MEMBER_ID ORDER BY MAP_DTE ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) LAST_GRADE_2,
        MAX(IFF(GRADE=3,MAP_DTE,NULL)) OVER(PARTITION BY MEMBER_ID ORDER BY MAP_DTE ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) LAST_GRADE_3,
        MAX(IFF(GRADE=4,MAP_DTE,NULL)) OVER(PARTITION BY MEMBER_ID ORDER BY MAP_DTE ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) LAST_GRADE_4
 FROM SLOT_SOURCE s
),
SLOT_FEATURES AS (
 SELECT MEMBER_ID,MAP_DTE,
        CASE WHEN GRADE IS NOT NULL THEN GRADE-LEAST(GRADE,COALESCE(CASE WHEN LAST_GRADE_1>=DATEADD('MONTH',-3,MAP_DTE) THEN 1 WHEN LAST_GRADE_2>=DATEADD('MONTH',-3,MAP_DTE) THEN 2 WHEN LAST_GRADE_3>=DATEADD('MONTH',-3,MAP_DTE) THEN 3 WHEN LAST_GRADE_4>=DATEADD('MONTH',-3,MAP_DTE) THEN 4 END, NULL)) END SLOT_F3,
        CASE WHEN GRADE IS NOT NULL THEN GRADE-LEAST(GRADE,COALESCE(CASE WHEN LAST_GRADE_1>=DATEADD('MONTH',-6,MAP_DTE) THEN 1 WHEN LAST_GRADE_2>=DATEADD('MONTH',-6,MAP_DTE) THEN 2 WHEN LAST_GRADE_3>=DATEADD('MONTH',-6,MAP_DTE) THEN 3 WHEN LAST_GRADE_4>=DATEADD('MONTH',-6,MAP_DTE) THEN 4 END, NULL)) END SLOT_F6,
        CASE WHEN GRADE IS NOT NULL THEN GRADE-LEAST(GRADE,COALESCE(CASE WHEN LAST_GRADE_1>=DATEADD('MONTH',-12,MAP_DTE) THEN 1 WHEN LAST_GRADE_2>=DATEADD('MONTH',-12,MAP_DTE) THEN 2 WHEN LAST_GRADE_3>=DATEADD('MONTH',-12,MAP_DTE) THEN 3 WHEN LAST_GRADE_4>=DATEADD('MONTH',-12,MAP_DTE) THEN 4 END, NULL)) END SLOT_F12
 FROM SLOT_HISTORY
),
-- Slotting-only mapping: missing unrelated members do not invalidate a group.
-- Require a single nonblank Head ID across all available member records.
-- Never infer Head ID from GEC_ID or resolve conflicting Head evidence.
SLOT_GROUP_KEYS AS (
 SELECT DISTINCT l.MAP_DTE,l.GEC_ID FROM LINK_OK l
 JOIN (SELECT DISTINCT MEMBER_ID FROM SLOT_SOURCE WHERE GRADE IS NOT NULL) s ON s.MEMBER_ID=l.MEMBER_ID
),
SLOT_GROUP_HEAD AS (
 SELECT l.MAP_DTE,l.GEC_ID,
        MIN(NULLIF(TRIM(TO_VARCHAR(t.GEC_HEAD_ID)),'')) HEAD_ID
 FROM SLOT_GROUP_KEYS k JOIN LINK_OK l ON l.MAP_DTE=k.MAP_DTE AND l.GEC_ID=k.GEC_ID
 JOIN {T} t ON l.MEMBER_ID=TRIM(TO_VARCHAR(t.ENTITY_ID)) AND l.MAP_DTE=TO_DATE(t.MAP_DTE)
 GROUP BY 1,2
 HAVING COUNT(DISTINCT NULLIF(TRIM(TO_VARCHAR(t.GEC_HEAD_ID)),''))=1
),
SLOT_HEAD_PRESENT AS (
 SELECT g.MAP_DTE,g.GEC_ID,g.HEAD_ID
 FROM SLOT_GROUP_HEAD g JOIN {T} ht ON TRIM(TO_VARCHAR(ht.ENTITY_ID))=g.HEAD_ID
 AND TO_DATE(ht.MAP_DTE)=g.MAP_DTE
 GROUP BY 1,2,3
 HAVING COUNT(*)=1
),
SLOT_MEMBER_RAW AS (
 SELECT h.MAP_DTE,h.HEAD_ID,l.MEMBER_ID,
        MAX(sf.SLOT_F3) SLOT_F3,MAX(sf.SLOT_F6) SLOT_F6,MAX(sf.SLOT_F12) SLOT_F12
 FROM SLOT_HEAD_PRESENT h JOIN LINK_OK l ON h.GEC_ID=l.GEC_ID AND h.MAP_DTE=l.MAP_DTE
 JOIN {T} t ON l.MEMBER_ID=TRIM(TO_VARCHAR(t.ENTITY_ID)) AND h.MAP_DTE=TO_DATE(t.MAP_DTE)
 JOIN (SELECT DISTINCT MEMBER_ID FROM SLOT_SOURCE WHERE GRADE IS NOT NULL) cohort ON cohort.MEMBER_ID=l.MEMBER_ID
 LEFT JOIN SLOT_FEATURES sf ON sf.MEMBER_ID=l.MEMBER_ID AND sf.MAP_DTE=h.MAP_DTE
 LEFT JOIN LINK_RAW hl ON hl.MEMBER_ID=h.HEAD_ID AND hl.MAP_DTE=h.MAP_DTE
 WHERE h.HEAD_ID<>l.MEMBER_ID AND h.MAP_DTE BETWEEN DATE '{start}' AND DATE '{end}'
 AND (hl.MEMBER_ID IS NULL OR (hl.N_GROUPS=1 AND hl.GEC_ID=h.GEC_ID))
 GROUP BY 1,2,3 HAVING COUNT(*)=1
),
MEMBER_RAW AS (
 SELECT h.MAP_DTE,h.HEAD_ID,l.MEMBER_ID,
        TRY_TO_DOUBLE(t.CLIENT_DOWNGRADE_BEST_3M::VARCHAR) F3,
        TRY_TO_DOUBLE(t.CLIENT_DOWNGRADE_BEST_6M::VARCHAR) F6,
        TRY_TO_DOUBLE(t.CLIENT_DOWNGRADE_BEST_12M::VARCHAR) F12,
        sf.SLOT_F3,sf.SLOT_F6,sf.SLOT_F12
 FROM HEAD_VERIFIED h JOIN LINK_OK l ON h.GEC_ID=l.GEC_ID AND h.MAP_DTE=l.MAP_DTE
 JOIN {T} t ON l.MEMBER_ID=TRIM(TO_VARCHAR(t.ENTITY_ID))
    AND h.MAP_DTE=TO_DATE(t.MAP_DTE)
 LEFT JOIN SLOT_FEATURES sf ON sf.MEMBER_ID=l.MEMBER_ID AND sf.MAP_DTE=h.MAP_DTE
 WHERE h.HEAD_ID<>l.MEMBER_ID
),
MEMBER_FEATURES AS (
 SELECT MAP_DTE,HEAD_ID,MEMBER_ID,
        'gec_member_rating_downgrade_best_3m' INDICATOR,F3 FEATURE FROM MEMBER_RAW
 UNION ALL
 SELECT MAP_DTE,HEAD_ID,MEMBER_ID,
        'gec_member_rating_downgrade_best_6m',F6 FROM MEMBER_RAW
 UNION ALL
 SELECT MAP_DTE,HEAD_ID,MEMBER_ID,
        'gec_member_rating_downgrade_best_12m',F12 FROM MEMBER_RAW
 UNION ALL
 SELECT MAP_DTE,HEAD_ID,MEMBER_ID,'gec_member_slotting_downgrade_best_3m',SLOT_F3 FROM SLOT_MEMBER_RAW
 UNION ALL
 SELECT MAP_DTE,HEAD_ID,MEMBER_ID,'gec_member_slotting_downgrade_best_6m',SLOT_F6 FROM SLOT_MEMBER_RAW
 UNION ALL
 SELECT MAP_DTE,HEAD_ID,MEMBER_ID,'gec_member_slotting_downgrade_best_12m',SLOT_F12 FROM SLOT_MEMBER_RAW
),
MEMBER_PREV AS (
 SELECT m.*,
   LAG(MAP_DTE) OVER(PARTITION BY HEAD_ID,MEMBER_ID,INDICATOR ORDER BY MAP_DTE) PREV_DATE,
   LAG(FEATURE) OVER(PARTITION BY HEAD_ID,MEMBER_ID,INDICATOR ORDER BY MAP_DTE) PREV_FEATURE
 FROM MEMBER_FEATURES m
),
GRID_RULES AS (
 SELECT COLUMN1::VARCHAR INDICATOR,COLUMN2::DOUBLE THRESHOLD_A,
        0::DOUBLE THRESHOLD_B FROM VALUES
 ('gec_member_rating_downgrade_best_3m',2),('gec_member_rating_downgrade_best_3m',3),('gec_member_rating_downgrade_best_3m',4),
 ('gec_member_rating_downgrade_best_6m',2),('gec_member_rating_downgrade_best_6m',3),('gec_member_rating_downgrade_best_6m',4),
 ('gec_member_rating_downgrade_best_12m',2),('gec_member_rating_downgrade_best_12m',3),('gec_member_rating_downgrade_best_12m',4),
 ('gec_member_rating_downgrade_best_12m',5),
 ('gec_member_slotting_downgrade_best_3m',1),
 ('gec_member_slotting_downgrade_best_3m',2),
 ('gec_member_slotting_downgrade_best_3m',3),
 ('gec_member_slotting_downgrade_best_6m',1),
 ('gec_member_slotting_downgrade_best_6m',2),
 ('gec_member_slotting_downgrade_best_6m',3),
 ('gec_member_slotting_downgrade_best_12m',1),
 ('gec_member_slotting_downgrade_best_12m',2),
 ('gec_member_slotting_downgrade_best_12m',3)
)
, SLOT_CROSS AS (
 SELECT DISTINCT m.HEAD_ID,m.MEMBER_ID,m.MAP_DTE,m.INDICATOR,g.THRESHOLD_A
 FROM MEMBER_PREV m JOIN GRID_RULES g ON g.INDICATOR=m.INDICATOR
 WHERE m.INDICATOR LIKE 'gec_member_slotting_%' AND m.FEATURE>=g.THRESHOLD_A
 AND m.PREV_FEATURE<g.THRESHOLD_A AND DATEDIFF('DAY',m.PREV_DATE,m.MAP_DTE) BETWEEN 1 AND 7
)
SELECT c.HEAD_ID AS HEAD_ENTITY_ID,c.MEMBER_ID AS MEMBER_ENTITY_ID,
       l.GEC_ID,c.MAP_DTE AS SIGNAL_DATE,m.PREV_DATE AS PREVIOUS_OBSERVATION_DATE,
       g.GRADE AS CURRENT_SLOTTING_GRADE,pg.GRADE AS PREVIOUS_OBSERVED_GRADE,
       g.GRADE-m.FEATURE AS BEST_COMPARISON_GRADE_12M,
       m.FEATURE AS DOWNGRADE_NOTCHES,m.PREV_FEATURE AS PREVIOUS_DOWNGRADE_NOTCHES
FROM SLOT_CROSS c
JOIN MEMBER_PREV m ON m.HEAD_ID=c.HEAD_ID AND m.MEMBER_ID=c.MEMBER_ID
  AND m.MAP_DTE=c.MAP_DTE AND m.INDICATOR=c.INDICATOR
LEFT JOIN LINK_OK l ON l.MEMBER_ID=c.MEMBER_ID AND l.MAP_DTE=c.MAP_DTE
LEFT JOIN SLOT_SOURCE g ON g.MEMBER_ID=c.MEMBER_ID AND g.MAP_DTE=c.MAP_DTE
LEFT JOIN SLOT_SOURCE pg ON pg.MEMBER_ID=c.MEMBER_ID AND pg.MAP_DTE=m.PREV_DATE
WHERE c.THRESHOLD_A=1 AND c.INDICATOR='gec_member_slotting_downgrade_best_12m'
ORDER BY c.HEAD_ID,c.MAP_DTE,c.MEMBER_ID
""".format(**params)
