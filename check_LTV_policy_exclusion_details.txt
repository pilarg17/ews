WITH SEG_RAW AS (
 SELECT TRIM(TO_VARCHAR(ENTITY_ID)) ENTITY_ID,TO_DATE(MAP_DTE) MAP_DTE,
        COUNT(*) N_ROWS,COUNT(DISTINCT UPPER(TRIM(TO_VARCHAR(CORPORATE_SEGMENT)))) N_SEG,
        MIN(UPPER(TRIM(TO_VARCHAR(CORPORATE_SEGMENT)))) SEGMENT
 FROM {E} WHERE ENTITY_ID IS NOT NULL AND MAP_DTE IS NOT NULL
 GROUP BY 1,2
),
SEG_EXACT AS (SELECT ENTITY_ID,MAP_DTE,SEGMENT FROM SEG_RAW WHERE N_ROWS=1 AND N_SEG=1),
NEED_HIST AS (
 SELECT DISTINCT TRIM(TO_VARCHAR(t.ENTITY_ID)) ENTITY_ID,TO_DATE(t.MAP_DTE) MAP_DTE
 FROM {T} t LEFT JOIN SEG_EXACT e ON e.ENTITY_ID=TRIM(TO_VARCHAR(t.ENTITY_ID))
  AND e.MAP_DTE=TO_DATE(t.MAP_DTE)
 WHERE e.ENTITY_ID IS NULL AND TO_DATE(t.MAP_DTE) BETWEEN DATE '{start}' AND DATE '{end}'
),
SEG_HISTORY AS (
 SELECT n.ENTITY_ID,n.MAP_DTE,COUNT(h.ENTITY_ID) N_MATCH,
        COUNT(DISTINCT UPPER(TRIM(TO_VARCHAR(h.CORPORATE_SEGMENT)))) N_SEG,
        MIN(UPPER(TRIM(TO_VARCHAR(h.CORPORATE_SEGMENT)))) SEGMENT
 FROM NEED_HIST n LEFT JOIN {H} h ON n.ENTITY_ID=TRIM(TO_VARCHAR(h.ENTITY_ID))
  AND n.MAP_DTE BETWEEN TO_DATE(h.TEK_STRT_DTE)
                    AND COALESCE(TO_DATE(h.TEK_END_DTE),DATE '9999-12-31')
 GROUP BY 1,2
),
SEG AS (
 SELECT ENTITY_ID,MAP_DTE,SEGMENT FROM SEG_EXACT
 UNION ALL
 SELECT ENTITY_ID,MAP_DTE,SEGMENT FROM SEG_HISTORY WHERE N_MATCH=1 AND N_SEG=1
),
ENTITY_BASE AS (
 SELECT TRIM(TO_VARCHAR(t.ENTITY_ID)) ENTITY_ID,TO_DATE(t.MAP_DTE) MAP_DTE,
        seg.SEGMENT,
        TRY_TO_NUMBER(t.TGT_ANY_SAMEDAY::VARCHAR) TARGET,
        IFF(TRY_TO_NUMBER(b.LOAN_COUNT::VARCHAR)>0 AND
            TRY_TO_NUMBER(b.LOAN_SOURCE_DUPLICATES::VARCHAR)=0 AND
            TRY_TO_NUMBER(b.NOMINAL_KNOWN_LOANS::VARCHAR)=TRY_TO_NUMBER(b.LOAN_COUNT::VARCHAR),
            TRY_TO_DOUBLE(b.NOMINAL_KNOWN_SUM::VARCHAR),NULL) EVENT_NOMINAL
 FROM {T} t JOIN {B} b ON TRIM(TO_VARCHAR(t.ENTITY_ID))=TRIM(TO_VARCHAR(b.ENTITY_ID))
    AND TO_DATE(t.MAP_DTE)=TO_DATE(b.MAP_DTE)
 LEFT JOIN SEG seg ON seg.ENTITY_ID=TRIM(TO_VARCHAR(t.ENTITY_ID))
    AND seg.MAP_DTE=TO_DATE(t.MAP_DTE)
 WHERE TO_DATE(t.MAP_DTE) BETWEEN DATE '{start}' AND DATE '{end}'
    AND TRY_TO_NUMBER(b.LOAN_COUNT::VARCHAR)>0
    AND TRY_TO_NUMBER(b.HAS_VALID_ENTITY_HISTORY::VARCHAR)=1
 QUALIFY ROW_NUMBER() OVER (PARTITION BY t.ENTITY_ID,t.MAP_DTE
                            ORDER BY b.MAP_DTE)=1
),
LOAN_CLASS AS (
 SELECT TRIM(TO_VARCHAR(ENTITY_ID)) ENTITY_ID,TO_DATE(MAP_DTE) MAP_DTE,
        TRIM(TO_VARCHAR(INSTRUMENT_ID)) INSTRUMENT_ID,
        MIN(TRY_TO_NUMBER(CONTRACT_INTERNAL_PURPOSE_CODE::VARCHAR)) PURPOSE,
        COUNT(DISTINCT TRY_TO_NUMBER(CONTRACT_INTERNAL_PURPOSE_CODE::VARCHAR)) N_PURPOSE,
        COUNT(*) N_ROWS
 FROM {L}
 WHERE TO_DATE(MAP_DTE) BETWEEN DATE '{start}' AND DATE '{end}'
 GROUP BY 1,2,3
),
LOAN_SEG AS (
 SELECT ENTITY_ID,MAP_DTE,INSTRUMENT_ID,
        CASE WHEN PURPOSE IN ({adc}) THEN 'ADC'
             WHEN PURPOSE IN ({ip}) THEN 'IP'
             WHEN PURPOSE IN ({other}) THEN 'OTHER_CORPORATE'
             ELSE NULL END PORTFOLIO
 FROM LOAN_CLASS WHERE N_PURPOSE=1 AND N_ROWS=1
),
SCOPE_RAW AS (
 SELECT 'GLOBAL' PORTFOLIO,p.ENTITY_ID,p.MAP_DTE,p.TARGET,p.EVENT_NOMINAL
 FROM ENTITY_BASE p
 UNION ALL
 SELECT p.SEGMENT,p.ENTITY_ID,p.MAP_DTE,p.TARGET,p.EVENT_NOMINAL
 FROM ENTITY_BASE p WHERE p.SEGMENT IN ('LARGE_CORPORATE','MID_CORPORATE')
 UNION ALL
 SELECT ls.PORTFOLIO,p.ENTITY_ID,p.MAP_DTE,p.TARGET,p.EVENT_NOMINAL
 FROM ENTITY_BASE p JOIN LOAN_SEG ls ON p.ENTITY_ID=ls.ENTITY_ID AND p.MAP_DTE=ls.MAP_DTE
 WHERE p.SEGMENT='OTHER_CORPORATE' AND ls.PORTFOLIO IS NOT NULL
),
SCOPE AS (
 SELECT PORTFOLIO,ENTITY_ID,MAP_DTE,MAX(TARGET) TARGET,
        MAX(EVENT_NOMINAL) EVENT_NOMINAL
 FROM SCOPE_RAW GROUP BY 1,2,3
 HAVING COUNT(DISTINCT COALESCE(TARGET,-1))=1
    AND COUNT(DISTINCT COALESCE(EVENT_NOMINAL,-1))=1
),
PORTS AS (
 SELECT COLUMN1::VARCHAR PORTFOLIO FROM VALUES
 ('GLOBAL'),('LARGE_CORPORATE'),('MID_CORPORATE'),('ADC'),('IP'),('OTHER_CORPORATE')
),
EVENT_FIRST AS (
 SELECT PORTFOLIO,ENTITY_ID,MIN(MAP_DTE) FIRST_OBS
 FROM SCOPE GROUP BY 1,2
),
EVENT_ACTIVE AS (
 SELECT p.*,LAG(MAP_DTE) OVER(PARTITION BY PORTFOLIO,ENTITY_ID ORDER BY MAP_DTE) PREV_ACTIVE
 FROM SCOPE p WHERE TARGET=1
),
EVENT_GROUP AS (
 SELECT a.*,SUM(IFF(PREV_ACTIVE IS NULL OR MAP_DTE>DATEADD('MONTH',3,PREV_ACTIVE),1,0))
         OVER(PARTITION BY PORTFOLIO,ENTITY_ID ORDER BY MAP_DTE ROWS UNBOUNDED PRECEDING) EVENT_GRP
 FROM EVENT_ACTIVE a
),
EVENT_START AS (
 SELECT PORTFOLIO,ENTITY_ID,MIN(MAP_DTE) EVENT_DATE
 FROM EVENT_GROUP GROUP BY 1,2,EVENT_GRP
),
EVENT_CANDIDATES AS (
 SELECT ev.PORTFOLIO,ev.ENTITY_ID,ev.EVENT_DATE,
        ev.PORTFOLIO||'|'||ev.ENTITY_ID||'|'||TO_VARCHAR(ev.EVENT_DATE) EVENT_ID,
        p.EVENT_NOMINAL
 FROM EVENT_START ev JOIN SCOPE p ON p.PORTFOLIO=ev.PORTFOLIO
    AND p.ENTITY_ID=ev.ENTITY_ID AND p.MAP_DTE=ev.EVENT_DATE
 JOIN EVENT_FIRST f ON f.PORTFOLIO=ev.PORTFOLIO AND f.ENTITY_ID=ev.ENTITY_ID
 WHERE ev.EVENT_DATE>=DATEADD('MONTH',11,f.FIRST_OBS)
   AND p.EVENT_NOMINAL>=50000
),
EVENTS AS (
 SELECT e.* FROM EVENT_CANDIDATES e
 WHERE EXISTS (SELECT 1 FROM SCOPE p WHERE p.PORTFOLIO=e.PORTFOLIO
  AND p.ENTITY_ID=e.ENTITY_ID
  AND p.MAP_DTE BETWEEN DATEADD('MONTH',-11,e.EVENT_DATE)
                    AND DATEADD('MONTH',-1,e.EVENT_DATE)
  AND p.TARGET=0 AND p.EVENT_NOMINAL>=50000)
),

LTV_RAW AS (
 SELECT TRIM(TO_VARCHAR(ENTITY_ID)) ENTITY_ID,
        TRIM(TO_VARCHAR(INSTRUMENT_ID)) INSTRUMENT_ID,
        TO_DATE(MAP_DTE) LTV_MONTH,
        TO_DATE(LTV_SNAPSHOT_DATE) VALUATION_DATE,
        TRY_TO_DOUBLE(LTV_RETAIL_METHOD::VARCHAR) LTV_VALUE,
        MAX(IFF(LTV_POLICY_STATUS='PROVISIONAL_MAPPING',TO_VARCHAR(LTV_POLICY_CATEGORY),NULL)) POLICY_CATEGORY,
        MAX(IFF(LTV_POLICY_STATUS='PROVISIONAL_MAPPING',TRY_TO_DOUBLE(LTV_POLICY_THRESHOLD::VARCHAR),NULL)) POLICY_THRESHOLD,
        COUNT(*) N_SOURCE_ROWS
 FROM {V}
 WHERE LTV_STATUS='CANDIDATE_RETAIL_METHOD' AND UPPER(TRIM(TO_VARCHAR(LOAN_CURRENCY)))='EUR'
 AND TRY_TO_DOUBLE(LTV_RETAIL_METHOD::VARCHAR)>0 AND TRY_TO_DOUBLE(ALLOCATED_COLLATERAL_VALUE::VARCHAR)>0
 AND COALESCE(TRY_TO_NUMBER(N_AMBIGUOUS_VALUATIONS::VARCHAR),0)=0
 AND COALESCE(TRY_TO_NUMBER(N_UNCLASSIFIED_LINKS::VARCHAR),0)=0
 AND COALESCE(TRY_TO_NUMBER(N_INCOMPLETE_REAL_ESTATE_LINKS::VARCHAR),0)=0
 AND COALESCE(TRY_TO_NUMBER(N_BAD_SHARED_EXPOSURES::VARCHAR),0)=0
 AND COALESCE(TRY_TO_NUMBER(N_DUPLICATE_LINKS::VARCHAR),0)=0
 AND TO_DATE(LTV_SNAPSHOT_DATE)<=TO_DATE(MAP_DTE)
 AND TO_DATE(PROTECTION_METRICS_DATE)<=TO_DATE(MAP_DTE)
 GROUP BY 1,2,3,4,5
),
LTV_OK AS (
 SELECT ENTITY_ID,INSTRUMENT_ID,LTV_MONTH,MIN(VALUATION_DATE) VALUATION_DATE,
        MAX(LTV_VALUE) LTV_VALUE,MAX(POLICY_CATEGORY) POLICY_CATEGORY,MAX(POLICY_THRESHOLD) POLICY_THRESHOLD
 FROM LTV_RAW GROUP BY 1,2,3
 HAVING SUM(N_SOURCE_ROWS)=1 AND COUNT(DISTINCT LTV_VALUE)=1
),
LTV_MONTHLY_UNIQUE AS (
 SELECT l.* FROM LTV_OK l
 QUALIFY COUNT(*) OVER(PARTITION BY ENTITY_ID,INSTRUMENT_ID,
     DATE_TRUNC('MONTH',LTV_MONTH))=1
),
LTV_DELAY AS (
 SELECT l.ENTITY_ID,l.INSTRUMENT_ID,l.LTV_MONTH,l.LTV_VALUE,l.POLICY_CATEGORY,l.POLICY_THRESHOLD,
        p.PORTFOLIO,p.MAP_DTE SIGNAL_DATE
 FROM LTV_MONTHLY_UNIQUE l LEFT JOIN LOAN_SEG ls ON ls.ENTITY_ID=l.ENTITY_ID
    AND ls.INSTRUMENT_ID=l.INSTRUMENT_ID AND ls.MAP_DTE=l.LTV_MONTH
 JOIN SCOPE p ON p.ENTITY_ID=l.ENTITY_ID
    AND p.MAP_DTE>l.LTV_MONTH AND p.MAP_DTE<=DATEADD('DAY',35,l.LTV_MONTH)
    AND (p.PORTFOLIO='GLOBAL' OR
         (p.PORTFOLIO IN ('LARGE_CORPORATE','MID_CORPORATE') AND
           p.PORTFOLIO IN (SELECT SEGMENT FROM ENTITY_BASE b WHERE b.ENTITY_ID=l.ENTITY_ID AND b.MAP_DTE=l.LTV_MONTH))
         OR p.PORTFOLIO=ls.PORTFOLIO)
 QUALIFY ROW_NUMBER() OVER(PARTITION BY p.PORTFOLIO,l.ENTITY_ID,l.INSTRUMENT_ID,l.LTV_MONTH
 ORDER BY p.MAP_DTE)=1
),
GRID_RULES AS (
 SELECT COLUMN1::VARCHAR INDICATOR,COLUMN2::DOUBLE THRESHOLD_A,
        COLUMN3::DOUBLE THRESHOLD_B FROM VALUES
 ('ltv_increase_15pp_1m',15,1),
 ('ltv_increase_15pp_3m',15,3),
 ('ltv_increase_15pp_6m',15,6),
 ('ltv_increase_15pp_12m',15,12),
 ('ltv_policy_breach',0,0),
 ('ltv_high_loan',0.8,0),('ltv_high_loan',1.0,0),('ltv_high_loan',1.2,0),
 ('ltv_high_and_client_rating_downgrade',0.8,2),('ltv_high_and_client_rating_downgrade',0.8,3),
 ('ltv_high_and_client_rating_downgrade',1.0,2),('ltv_high_and_client_rating_downgrade',1.0,3),
 ('ltv_high_and_client_rating_downgrade',1.2,2),('ltv_high_and_client_rating_downgrade',1.2,3)
),
LTV_FEATURES AS (
 SELECT m.PORTFOLIO,m.ENTITY_ID,m.INSTRUMENT_ID,m.SIGNAL_DATE,m.LTV_MONTH,
        g.INDICATOR,g.THRESHOLD_A,g.THRESHOLD_B,
        m.LTV_VALUE,p.LTV_VALUE PREV_LTV,
        ROUND(100.0*(m.LTV_VALUE-p.LTV_VALUE),10) LTV_INCREASE_PP
 FROM LTV_DELAY m CROSS JOIN GRID_RULES g
 JOIN LTV_MONTHLY_UNIQUE p ON p.ENTITY_ID=m.ENTITY_ID
    AND p.INSTRUMENT_ID=m.INSTRUMENT_ID
    AND DATE_TRUNC('MONTH',p.LTV_MONTH)=DATEADD('MONTH',-CAST(g.THRESHOLD_B AS INTEGER),DATE_TRUNC('MONTH',m.LTV_MONTH))
 WHERE g.INDICATOR LIKE 'ltv_increase_15pp_%'
),
RATING_DELAY AS (
 SELECT l.*,TRY_TO_DOUBLE(t.CLIENT_DOWNGRADE_BEST_3M::VARCHAR) RATING_DOWNGRADE_3M
 FROM LTV_DELAY l JOIN {T} t ON TRIM(TO_VARCHAR(t.ENTITY_ID))=l.ENTITY_ID
   AND TO_DATE(t.MAP_DTE)=l.SIGNAL_DATE
 QUALIFY ROW_NUMBER() OVER(PARTITION BY l.PORTFOLIO,l.ENTITY_ID,l.INSTRUMENT_ID,l.LTV_MONTH
   ORDER BY t.MAP_DTE)=1
),
REFERENCE_FEATURES AS (
 SELECT p.*,
   LAG(SIGNAL_DATE) OVER(PARTITION BY PORTFOLIO,ENTITY_ID,INSTRUMENT_ID ORDER BY LTV_MONTH) PREV_DATE,
   LAG(LTV_VALUE) OVER(PARTITION BY PORTFOLIO,ENTITY_ID,INSTRUMENT_ID ORDER BY LTV_MONTH) PREV_LTV,
   LAG(RATING_DOWNGRADE_3M) OVER(PARTITION BY PORTFOLIO,ENTITY_ID,INSTRUMENT_ID ORDER BY LTV_MONTH) PREV_RATING_DOWNGRADE_3M
 FROM RATING_DELAY p
),
POLICY_FEATURES AS (
 SELECT m.PORTFOLIO,m.ENTITY_ID,m.INSTRUMENT_ID,m.SIGNAL_DATE,
        m.LTV_VALUE,m.POLICY_THRESHOLD,p.LTV_VALUE PREV_LTV,p.POLICY_THRESHOLD PREV_POLICY_THRESHOLD
 FROM LTV_DELAY m JOIN LTV_MONTHLY_UNIQUE p ON p.ENTITY_ID=m.ENTITY_ID
   AND p.INSTRUMENT_ID=m.INSTRUMENT_ID
   AND DATE_TRUNC('MONTH',p.LTV_MONTH)=DATEADD('MONTH',-1,DATE_TRUNC('MONTH',m.LTV_MONTH))
 WHERE m.PORTFOLIO IN ('GLOBAL','ADC')
   AND m.POLICY_CATEGORY=p.POLICY_CATEGORY AND m.POLICY_THRESHOLD=p.POLICY_THRESHOLD
   AND m.POLICY_THRESHOLD>0 AND p.POLICY_THRESHOLD>0
)
, POLICY_CROSS AS (
 SELECT DISTINCT PORTFOLIO,ENTITY_ID,INSTRUMENT_ID,SIGNAL_DATE
 FROM POLICY_FEATURES WHERE PREV_LTV<=PREV_POLICY_THRESHOLD AND LTV_VALUE>POLICY_THRESHOLD
), CROSS_QA AS (
 SELECT c.*,p.TARGET,p.EVENT_NOMINAL,
 CASE WHEN p.ENTITY_ID IS NULL THEN 'OUTSIDE_SCOPE'
 WHEN p.TARGET IS NULL THEN 'TARGET_UNKNOWN'
 WHEN p.TARGET<>0 THEN 'TARGET_ALREADY_ACTIVE'
 WHEN p.EVENT_NOMINAL IS NULL THEN 'EXPOSURE_UNKNOWN_OR_INCOMPLETE'
 WHEN p.EVENT_NOMINAL<50000 THEN 'EXPOSURE_BELOW_50K'
 WHEN c.SIGNAL_DATE>DATEADD('MONTH',-1,DATE '{end}') THEN 'RIGHT_CENSORED'
 ELSE 'ELIGIBLE' END REASON
 FROM POLICY_CROSS c LEFT JOIN SCOPE p ON p.PORTFOLIO=c.PORTFOLIO AND p.ENTITY_ID=c.ENTITY_ID AND p.MAP_DTE=c.SIGNAL_DATE
)
SELECT DISTINCT c.REASON,
 DENSE_RANK() OVER (ORDER BY c.ENTITY_ID) AS CLIENT_NO,
 DENSE_RANK() OVER (ORDER BY c.ENTITY_ID,c.INSTRUMENT_ID) AS LOAN_NO,
 c.SIGNAL_DATE,
 TO_VARCHAR(t.TGT_ANY_SAMEDAY) AS TARGET_RAW,
 TO_VARCHAR(b.PD_CRR) AS PD_CRR,
 TO_VARCHAR(b.DPD_MAX) AS DPD_MAX,
 TO_VARCHAR(t.TGT_DEFAULT_SAMEDAY) AS TGT_DEFAULT,
 TO_VARCHAR(t.TGT_FORBEARANCE_SAMEDAY) AS TGT_FORBEARANCE,
 TO_VARCHAR(t.TGT_DIM1_SAMEDAY) AS TGT_DIM1,
 TO_VARCHAR(b.LOAN_COUNT) AS LOAN_COUNT,
 TO_VARCHAR(b.NOMINAL_KNOWN_LOANS) AS NOMINAL_KNOWN_LOANS,
 TO_VARCHAR(b.NOMINAL_KNOWN_SUM) AS NOMINAL_KNOWN_SUM,
 TO_VARCHAR(b.LOAN_SOURCE_DUPLICATES) AS LOAN_SOURCE_DUPLICATES,
 c.EVENT_NOMINAL,
 CASE WHEN c.REASON='EXPOSURE_UNKNOWN_OR_INCOMPLETE' THEN
   CASE WHEN TRY_TO_NUMBER(b.LOAN_SOURCE_DUPLICATES::VARCHAR) IS NULL THEN 'DUPLICATE_CHECK_UNKNOWN'
        WHEN TRY_TO_NUMBER(b.LOAN_SOURCE_DUPLICATES::VARCHAR)<>0 THEN 'DUPLICATE_LOAN_SOURCE'
        WHEN TRY_TO_NUMBER(b.NOMINAL_KNOWN_LOANS::VARCHAR) IS NULL THEN 'KNOWN_LOAN_COUNT_UNKNOWN'
        WHEN TRY_TO_NUMBER(b.NOMINAL_KNOWN_LOANS::VARCHAR)<>TRY_TO_NUMBER(b.LOAN_COUNT::VARCHAR) THEN 'INCOMPLETE_NOMINAL_COVERAGE'
        WHEN TRY_TO_DOUBLE(b.NOMINAL_KNOWN_SUM::VARCHAR) IS NULL THEN 'NOMINAL_SUM_UNKNOWN'
        ELSE 'OTHER_OR_CONFLICTING_SOURCE' END
 END AS EXPOSURE_DETAIL
 FROM CROSS_QA c
 LEFT JOIN {T} t ON TRIM(TO_VARCHAR(t.ENTITY_ID))=c.ENTITY_ID AND TO_DATE(t.MAP_DTE)=c.SIGNAL_DATE
 LEFT JOIN {B} b ON TRIM(TO_VARCHAR(b.ENTITY_ID))=c.ENTITY_ID AND TO_DATE(b.MAP_DTE)=c.SIGNAL_DATE
 WHERE c.PORTFOLIO='ADC' AND c.REASON IN ('TARGET_UNKNOWN','EXPOSURE_UNKNOWN_OR_INCOMPLETE')
 ORDER BY c.REASON,CLIENT_NO,LOAN_NO,c.SIGNAL_DATE
""".format(**params)
