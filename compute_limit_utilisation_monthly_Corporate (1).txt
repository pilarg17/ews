def log(message):
    print('[%7.1fs] %s' % (time.monotonic()-T0, message), flush=True)

def show(title, x, cap=80):
    print('\n' + title, flush=True)
    if isinstance(x, (list, tuple, dict)):
        x = pd.DataFrame([x] if isinstance(x, dict) else x)
    print('NO ROWS' if x is None or x.empty else
          x.head(cap).to_csv(index=False, sep=';', na_rep='NULL'), flush=True)

def norm(v):
    if v is None or pd.isna(v):
        return None
    s = str(v).strip()
    if not s or s.lower() in ('nan', 'none', 'null'):
        return None
    return re.sub(r'\.0+$', '', s) if re.fullmatch(r'[0-9]+\.0+', s) else s

def safe_ident(value):
    value = str(value).replace('${projectKey}', PROJECT)
    if not re.fullmatch(r'[A-Za-z_][A-Za-z0-9_$]*', value):
        raise RuntimeError('Unexpected table identifier')
    return '"' + value + '"'

def physical(dataset, required):
    project = dataiku.api_client().get_project(PROJECT)
    params = project.get_dataset(dataset).get_settings().get_raw().get('params') or {}
    if params.get('connection') != SNOW:
        raise RuntimeError('Unexpected Snowflake source for ' + dataset)
    present = {str(c['name']).upper() for c in dataiku.Dataset(dataset,project_key=PROJECT).read_schema()}
    if not set(required).issubset(present):
        raise RuntimeError('Missing required columns in ' + dataset + ': ' + str(sorted(set(required)-present)))
    return safe_ident(params['schema']) + '.' + safe_ident(params['table'])

def query(connection, title, statement, max_rows):
    if not re.match(r'^\s*(SELECT|WITH)\b', statement, re.I):
        raise RuntimeError('Only read-only SQL permitted')
    log('QUERY START | ' + title)
    begin = time.monotonic()
    try:
        x = SQLExecutor2(connection=connection).query_to_df(statement)
    except Exception as exc:
        log('QUERY FAILED | %s | %s' % (title, type(exc).__name__))
        raise RuntimeError('Read-only source query failed: ' + title) from None
    if len(x) > max_rows:
        raise RuntimeError('Source query exceeds safety row limit for ' + title)
    x.columns = [str(c).upper() for c in x.columns]
    log('QUERY DONE | %s | %d rows | %.1fs' % (title,len(x),time.monotonic()-begin))
    return x

def chunks(values):
    # Dataiku/Oracle may return source references as integer or float. Always
    # canonicalize identifiers before sorting and building any Oracle IN list.
    values = sorted(set(v for v in (norm(x) for x in values)
                        if v is not None and len(v) <= 100
                        and all(ord(ch) >= 32 for ch in v)))
    for k in range(0, len(values), MAX_BATCH_KEYS):
        yield values[k:k + MAX_BATCH_KEYS]

def ora_in(values):
    values = [norm(v) for v in values]
    if (not values or len(values) > MAX_BATCH_KEYS or
        any(v is None or len(v) > 100 or any(ord(ch) < 32 for ch in v)
            for v in values)):
        raise RuntimeError('Invalid Oracle key batch')
    return '(' + ','.join("'" + v.replace("'", "''") + "'" for v in values) + ')'

def source_text(v, marker):
    # Oracle SQL prefixes identifiers, preventing drivers from inferring numeric
    # and losing leading zeros / precision on long reference IDs.
    raw = norm(v)
    if raw is None:
        return None
    if not raw.startswith(marker):
        raise RuntimeError('Unexpected source identifier representation')
    return norm(raw[len(marker):])

def as_date(s):
    v = norm(s)
    if not v:
        return pd.NaT
    if re.fullmatch(r'\d{8}',v):
        return pd.to_datetime(v,format='%Y%m%d',errors='coerce')
    if re.fullmatch(r'\d{4}-\d{2}-\d{2}',v):
        return pd.to_datetime(v,format='%Y-%m-%d',errors='coerce')
    return pd.NaT

def after211(s):
    s = norm(s)
    return s.split('211',1)[1] if s and '211' in s and s.split('211',1)[1] else None

            # thousands of old report versions for each source reference.

                "SELECT 'R|'||TRIM(TO_CHAR({field})) LOAN_REF, "
                "TRIM(TO_CHAR(REPORT_DATEKEY)) REPORT_DAY, "
                "'V|'||TRIM(TO_CHAR(REPORT_VERSIONID)) REPORT_VERSION, "
                "ROW_NUMBER() OVER (PARTITION BY TRIM(TO_CHAR({field})) "
                "ORDER BY REPLACE(TRIM(TO_CHAR(REPORT_DATEKEY)),'-','') DESC, "
                "CASE WHEN REGEXP_LIKE(TRIM(TO_CHAR(REPORT_VERSIONID)), "
                "'^[0-9]+([.]0+)?$') THEN TO_NUMBER(REPORT_VERSIONID) "
                "END DESC NULLS LAST) AS RN "
                "FROM DTM_RFO_PLY.{view} WHERE {date_pred} "
                "AND TRIM(TO_CHAR({field})) IN {key_values} "
                ") WHERE RN=1").format(field=field,view=view,date_pred=date_pred,
                                         key_values=ora_in(batch))
            frame=query(ORACLE,'RFO %s %s BATCH %d'%(view,ym,batch_no+1),
                        rfo_query,MAX_BATCH_KEYS+10)
            rfo_parts.append(frame)
    rfo=pd.concat(rfo_parts,ignore_index=True) if rfo_parts else pd.DataFrame(columns=['LOAN_REF','REPORT_DAY','REPORT_VERSION'])
    if len(rfo):
        # This is essential: RFO LOAN_REF can arrive as int64 from Dataiku,
        # while INSTRUMENT_ID from the loan dataset is a string.
        rfo['LOAN_REF']=rfo['LOAN_REF'].map(lambda v: source_text(v, 'R|'))

# Engineering is computed directly from in-memory monthly history;
# no second Oracle or Snowflake read is necessary in this combined recipe.
THRESHOLDS_PCT = tuple(range(70, 101))  # 1pp grid, configurable
WINDOWS = (3, 6, 9, 12)

def compute_entity(group):
    entity = group['ENTITY_ID'].iloc[0]
    group = group.sort_values('LIMIT_MONTH')
    # REINDEX creates explicit null months, preventing false consecutive runs.
    idx = pd.date_range(group['LIMIT_MONTH'].min(),group['LIMIT_MONTH'].max(),freq='MS')
    v = group.set_index('LIMIT_MONTH').reindex(idx)
    v.index.name = 'LIMIT_MONTH'
    present = v['ENTITY_ID'].notna()
    v['ENTITY_ID']=entity
    valid = v['VALID_RATIO'].fillna(False).astype(bool)
    u = pd.to_numeric(v['UTILISATION_PCT'],errors='coerce').where(valid)
    v['UTILISATION_PCT'] = u
    base = u.shift(12)
    v['UTILISATION_12M_AGO_PCT'] = base
    v['CHANGE_12M_PP'] = u-base
    v['CHANGE_12M_REL_PCT'] = 100.0*(u-base)/base.where(base>0)
    for n in WINDOWS:
        prior_max = u.shift(1).rolling(n,min_periods=n).max()
        v['PRIOR_%dM_MAX_PCT'%n] = prior_max
        v['CHANGE_VS_PRIOR_%dM_MAX_PP'%n] = u-prior_max
        v['CHANGE_VS_PRIOR_%dM_MAX_REL_PCT'%n] = 100.0*(u-prior_max)/prior_max.where(prior_max>0)
        # Monthly approximation ONLY: weights are observed calendar days up to
        # as-of snapshot, not actual durations of intra-month utilisation.
        dt = pd.to_datetime(v['AS_OF_DATE'],errors='coerce')
        days = dt.dt.day.astype('float64').where(valid)
        numerator = (u*days).rolling(n,min_periods=n).sum()
        denominator = days.rolling(n,min_periods=n).sum()
        v['WEIGHTED_MONTHLY_PROXY_%dM_PCT'%n] = numerator/denominator.where(denominator>0)
    values = u.to_numpy(dtype=float)
    for t in THRESHOLDS_PCT:
        streak = np.zeros(len(v),dtype=np.int16)
        proven = np.zeros(len(v),dtype=bool)
        current_run = 0
        known_start = False
        seen_below = False
        for i, x in enumerate(values):
            if not np.isfinite(x):
                current_run = 0
                known_start = False
                seen_below = False
            elif x < t:
                current_run = 0
                known_start = False
                seen_below = True
            else:
                if current_run == 0:
                    known_start = seen_below
                current_run += 1
                streak[i] = current_run
                proven[i] = known_start
                seen_below = False
        v['STREAK_GE_%d_MONTHS'%t]=streak
        v['STREAK_GE_%d_START_KNOWN'%t]=proven
    # Baseline crossing classification only for the old 5 threshold levels.
    prior = u.shift(1)
    for t in (70,80,90,95,100):
        v['CROSSED_GE_%d'%t] = (u.ge(t)&prior.lt(t)).fillna(False)
    v['LIMIT_MONTH'] = v.index
    return v.loc[present].reset_index(drop=True)


def build_features(z):
    z = z.copy()
    needed={'ENTITY_ID','MONTH','AS_OF_DATE','UTILISATION_PCT','VALID_RATIO'}
    if not needed.issubset(z.columns):
        raise RuntimeError('Missing source history fields')
    z['ENTITY_ID'] = z['ENTITY_ID'].astype(str).str.strip()
    z['LIMIT_MONTH'] = pd.to_datetime(z['MONTH'].astype(str).str[:7]+'-01',errors='coerce')
    z['AS_OF_DATE'] = pd.to_datetime(z['AS_OF_DATE'],errors='coerce',utc=True).dt.tz_localize(None)
    z['UTILISATION_PCT'] = pd.to_numeric(z['UTILISATION_PCT'],errors='coerce')
    z['VALID_RATIO'] = z['VALID_RATIO'].fillna(False).astype(bool)
    if z[['ENTITY_ID','LIMIT_MONTH']].isna().any().any() or z['ENTITY_ID'].eq('').any():
        raise RuntimeError('Invalid entity/month')
    if z.duplicated(['ENTITY_ID','LIMIT_MONTH']).any():
        raise RuntimeError('Duplicate entity/month')
    if z.loc[z['VALID_RATIO'],'UTILISATION_PCT'].isna().any():
        raise RuntimeError('Valid ratio has missing utilisation')
    parts=[]
    for n,(_,g) in enumerate(z.groupby('ENTITY_ID',sort=False),1):
        parts.append(compute_entity(g))
        if n%2500==0: log('FEATURE GROUPS | entities=%d'%n)
    if not parts: raise RuntimeError('Empty feature output')
    out=pd.concat(parts,ignore_index=True)
    if len(out)!=len(z) or out.duplicated(['ENTITY_ID','LIMIT_MONTH']).any():
        raise RuntimeError('Feature rows/keys do not reconcile with monthly source')
    if out['UTILISATION_PCT'].notna().sum() != z['VALID_RATIO'].sum():
        raise RuntimeError('Feature eligibility does not reconcile with source')
    core=['ENTITY_ID','MONTH','LIMIT_MONTH','AS_OF_DATE','VALID_RATIO',
          'UTILISATION_PCT','LIMIT_SOURCE_DATE','LIMIT_SOURCE_LAG_DAYS',
          'DRAWN_EUR','LIMIT_EUR','HAS_SHARED_LIMIT','ALL_ROWS_COMPLETE',
          'UTILISATION_12M_AGO_PCT','CHANGE_12M_PP','CHANGE_12M_REL_PCT']
    other=[]
    for win in WINDOWS:
        other += ['PRIOR_%dM_MAX_PCT'%win,'CHANGE_VS_PRIOR_%dM_MAX_PP'%win,
                  'CHANGE_VS_PRIOR_%dM_MAX_REL_PCT'%win,
                  'WEIGHTED_MONTHLY_PROXY_%dM_PCT'%win]
    for t in THRESHOLDS_PCT:
        other += ['STREAK_GE_%d_MONTHS'%t,'STREAK_GE_%d_START_KNOWN'%t]
    other += ['CROSSED_GE_%d'%t for t in (70,80,90,95,100)]
    out=out[[c for c in core if c in out.columns]+other].copy()
    out=out.replace([np.inf,-np.inf],np.nan)
    log('FEATURES QA | change12=%d | prior3=%d | prior12=%d | weighted12_proxy=%d' % (
        int(out['CHANGE_12M_PP'].notna().sum()),
        int(out['PRIOR_3M_MAX_PCT'].notna().sum()),
        int(out['PRIOR_12M_MAX_PCT'].notna().sum()),
        int(out['WEIGHTED_MONTHLY_PROXY_12M_PCT'].notna().sum())))
    return out

log('FEATURE GENERATION START | from full reconstructed monthly history')
